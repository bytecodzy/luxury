   367→   367→function rgbToColorName(r: number, g: number, b: number): string {
   368→   368→  // Convert RGB to HSV for better color naming
   369→   369→  const rn = r / 255, gn = g / 255, bn = b / 255
   370→   370→  const max = Math.max(rn, gn, bn)
   371→   371→  const min = Math.min(rn, gn, bn)
   372→   372→  const delta = max - min
   373→   373→  const v = max
   374→   374→  const s = max === 0 ? 0 : delta / max
   375→   375→  let h = 0
   376→   376→  if (delta !== 0) {
   377→   377→    if (max === rn) h = ((gn - bn) / delta) % 6
   378→   378→    else if (max === gn) h = (bn - rn) / delta + 2
   379→   379→    else h = (rn - gn) / delta + 4
   380→   380→    h *= 60
   381→   381→    if (h < 0) h += 360
   382→   382→  }
   383→   383→
   384→   384→  // Achromatic (grey/black/white)
   385→   385→  if (s < 0.12) {
   386→   386→    if (v < 0.15) return 'black'
   387→   387→    if (v > 0.92) return 'white'
   388→   388→    if (v < 0.4) return 'charcoal grey'
   389→   389→    if (v < 0.65) return 'grey'
   390→   390→    return 'silver'
   391→   391→  }
   392→   392→
   393→   393→  // Bright/light/dark prefix — generous with "bright" for saturated colours
   394→   394→  // so FLUX generates vibrant results (product photos are usually well-lit)
   395→   395→  const lightPrefix = v > 0.6 ? 'bright ' : v < 0.25 ? 'dark ' : ''
   396→   396→
   397→   397→  // Hue-based naming
   398→   398→  if (h < 15 || h >= 345) return `${lightPrefix}red`
   399→   399→  if (h < 30) return v < 0.4 ? 'maroon' : 'red'
   400→   400→  if (h < 45) return v < 0.4 ? 'burgundy' : 'orange-red'
   401→   401→  if (h < 60) return `${lightPrefix}orange`
   402→   402→  if (h < 70) return 'mustard yellow'
   403→   403→  if (h < 85) return `${lightPrefix}yellow`
   404→   404→  if (h < 100) return v > 0.6 ? 'lime' : 'olive'
   405→   405→  if (h < 150) return `${lightPrefix}green`
   406→   406→  if (h < 175) return 'emerald green'
   407→   407→  if (h < 195) return 'teal'
   408→   408→  if (h < 215) return 'turquoise'
   409→   409→  if (h < 240) return `${lightPrefix}blue`
   410→   410→  if (h < 260) return 'navy blue'
   411→   411→  if (h < 285) return 'violet'
   412→   412→  if (h < 310) return v > 0.6 ? 'pink' : 'purple'
   413→   413→  if (h < 335) return v > 0.7 ? 'rose pink' : 'magenta'
   414→   414→  return `${lightPrefix}red`
   415→   415→}

   417→   417→export async function extractColorsFromProductImage(imageBase64: string): Promise<string> {
   418→   418→  if (!imageBase64 || !imageBase64.startsWith('data:image/')) return ''
   419→   419→
   420→   420→  try {
   421→   421→    const raw = stripDataUrl(imageBase64)
   422→   422→    const buf = Buffer.from(raw, 'base64')
   423→   423→
   424→   424→    // Use Jimp (pure-JS, no native binary — works on Vercel AND local)
   425→   425→    const JimpModule = await import('jimp')
   426→   426→    const Jimp = (JimpModule as any).Jimp || (JimpModule as any).default || JimpModule
   427→   427→    const image = await Jimp.read(buf)
   428→   428→    image.resize({ w: 32, h: 32 })
   429→   429→
   430→   430→    // Jimp's bitmap.data is RGBA (4 bytes per pixel)
   431→   431→    const data = image.bitmap.data as Buffer
   432→   432→    const pixelCount = 32 * 32
   433→   433→
   434→   434→    // Quantize pixels into colour buckets, weighting by SATURATION
   435→   435→    // (vibrant product colours get higher priority than grey backgrounds)
   436→   436→    const buckets = new Map<string, { count: number; satSum: number; r: number; g: number; b: number }>()
   437→   437→    for (let i = 0; i < pixelCount; i++) {
   438→   438→      const offset = i * 4
   439→   439→      const r = data[offset], g = data[offset + 1], b = data[offset + 2]
   440→   440→      // Skip transparent pixels
   441→   441→      if (data[offset + 3] < 128) continue
   442→   442→
   443→   443→      const max = Math.max(r, g, b), min = Math.min(r, g, b)
   444→   444→      const delta = max - min
   445→   445→      const sat = max === 0 ? 0 : delta / max // 0..1
   446→   446→
   447→   447→      // Skip near-white backgrounds (common in e-commerce product photos)
   448→   448→      if (max > 235 && delta < 15) continue
   449→   449→      // Skip near-black
   450→   450→      if (max < 25) continue
   451→   451→      // Skip low-saturation greys (background, shadows, mannequin)
   452→   452→      // Only keep pixels with saturation > 0.18 (vibrant enough to be product colour)
   453→   453→      if (sat < 0.18) continue
   454→   454→
   455→   455→      // Quantize to 3 bits per channel
   456→   456→      const key = `${r >> 5}-${g >> 5}-${b >> 5}`
   457→   457→      const existing = buckets.get(key)
   458→   458→      if (existing) {
   459→   459→        existing.count++
   460→   460→        existing.satSum += sat
   461→   461→        existing.r += r
   462→   462→        existing.g += g
   463→   463→        existing.b += b
   464→   464→      } else {
   465→   465→        buckets.set(key, { count: 1, satSum: sat, r, g, b })
   466→   466→      }
   467→   467→    }
   468→   468→
   469→   469→    if (buckets.size === 0) {
   470→   470→      // Fallback: if no vibrant colours found (e.g. black/white/silver product),
   471→   471→      // re-run without the saturation filter
   472→   472→      const fallback = new Map<string, { count: number; r: number; g: number; b: number }>()
   473→   473→      for (let i = 0; i < pixelCount; i++) {
   474→   474→        const offset = i * 4
   475→   475→        const r = data[offset], g = data[offset + 1], b = data[offset + 2]
   476→   476→        if (data[offset + 3] < 128) continue
   477→   477→        const max = Math.max(r, g, b), min = Math.min(r, g, b)
   478→   478→        if (max > 235 && max - min < 15) continue
   479→   479→        if (max < 25) continue
   480→   480→        const key = `${r >> 5}-${g >> 5}-${b >> 5}`
   481→   481→        const existing = fallback.get(key)
   482→   482→        if (existing) { existing.count++; existing.r += r; existing.g += g; existing.b += b }
   483→   483→        else fallback.set(key, { count: 1, r, g, b })
   484→   484→      }
   485→   485→      if (fallback.size === 0) return ''
   486→   486→      const sorted = Array.from(fallback.values()).sort((a, b) => b.count - a.count)
   487→   487→      const names = sorted.slice(0, 2).map(bk => rgbToColorName(Math.round(bk.r / bk.count), Math.round(bk.g / bk.count), Math.round(bk.b / bk.count)))
   488→   488→      const unique = Array.from(new Set(names))
   489→   489→      console.log(`[virtual-tryon] Image-extracted colours (fallback): ${unique.join(', ')}`)
   490→   490→      return unique.join(', ')
   491→   491→    }
   492→   492→
   493→   493→    // Sort by (count × avg saturation) — vibrant + frequent colours win
   494→   494→    const sorted = Array.from(buckets.values()).sort((a, b) =>
   495→   495→      (b.count * (b.satSum / b.count)) - (a.count * (a.satSum / a.count))
   496→   496→    )
   497→   497→    const top = sorted.slice(0, 4)
   498→   498→    const names = top.map(bucket => {
   499→   499→      const avgR = Math.round(bucket.r / bucket.count)
   500→   500→      const avgG = Math.round(bucket.g / bucket.count)
   501→   501→      const avgB = Math.round(bucket.b / bucket.count)
   502→   502→      return rgbToColorName(avgR, avgG, avgB)
   503→   503→    })
   504→   504→
   505→   505→    // Deduplicate by BASE colour name — if "red" and "bright red" both appear,
   506→   506→    // keep only "bright red" (the more descriptive/vibrant variant). This
   507→   507→    // prevents FLUX from averaging two reds into a muted medium-red.
   508→   508→    const byBase = new Map<string, string>()
   509→   509→    for (const name of names) {
   510→   510→      const base = name.replace(/^(bright |dark )/, '').trim()
   511→   511→      const existing = byBase.get(base)
   512→   512→      // Prefer "bright" variant over plain, and plain over "dark"
   513→   513→      if (!existing) {
   514→   514→        byBase.set(base, name)
   515→   515→      } else if (name.startsWith('bright ') && !existing.startsWith('bright ')) {
   516→   516→        byBase.set(base, name) // upgrade to bright
   517→   517→      }
   518→   518→    }
   519→   519→    // Preserve original frequency order
   520→   520→    const unique: string[] = []
   521→   521→    for (const name of names) {
   522→   522→      const base = name.replace(/^(bright |dark )/, '').trim()
   523→   523→      const chosen = byBase.get(base)
   524→   524→      if (chosen === name && !unique.includes(name)) {
   525→   525→        unique.push(name)
   526→   526→        byBase.delete(base) // only add once
   527→   527→      }
   528→   528→      if (unique.length >= 2) break // max 2 colours to keep the prompt focused
   529→   529→    }
   530→   530→    console.log(`[virtual-tryon] Image-extracted colours: ${unique.join(', ')} (from ${buckets.size} vibrant buckets)`)
   531→   531→    return unique.join(', ')
   532→   532→  } catch (err) {
   533→   533→    console.log(`[virtual-tryon] Colour extraction failed: ${err instanceof Error ? err.message : String(err)}`)
   534→   534→    return ''
   535→   535→  }
   536→   536→}

   595→   595→        size: config.size,

   614→   614→      else if (head.startsWith('89504e47')) mime = 'image/png'
   615→   615→      else if (head.startsWith('52494646')) mime = 'image/webp'
   616→   616→    }

   622→   622→  } catch (err) {

   624→   624→    console.log(`[virtual-tryon] ZAI ${strategyName} failed: ${msg.substring(0, 200)}`)
   625→   625→    return { success: false, error: msg.substring(0, 200) }
   626→   626→  }
   627→   627→}

   636→   636→async function uploadToTmpfiles(buf: Buffer, timeoutMs: number): Promise<string | null> {
   637→   637→  const boundary = '----3boxesTryon' + Math.random().toString(16).slice(2)
   638→   638→  const header = Buffer.from(
   639→   639→    `--${boundary}\r\nContent-Disposition: form-data; name="file"; filename="image.jpg"\r\nContent-Type: image/jpeg\r\n\r\n`
   640→   640→  )

   642→   642→  const body = Buffer.concat([header, buf, footer])
   643→   643→
   644→   644→  const controller = new AbortController()
   645→   645→  const timeoutId = setTimeout(() => controller.abort(), timeoutMs)
   646→   646→  try {
   647→   647→    const res = await fetch('https://tmpfiles.org/api/v1/upload', {
   648→   648→      method: 'POST',
   649→   649→      headers: {
   650→   650→        'Content-Type': `multipart/form-data; boundary=${boundary}`,
   651→   651→        'User-Agent': '3BOXES-TryOn/1.0',
   652→   652→      },
   653→   653→      body,
   654→   654→      signal: controller.signal,
   655→   655→    })
   656→   656→    clearTimeout(timeoutId)
   657→   657→    if (!res.ok) return null
   658→   658→    const json = (await res.json()) as { data?: { url?: string } }
   659→   659→    const viewerUrl = json?.data?.url
   660→   660→    if (!viewerUrl) return null
   661→   661→    return viewerUrl.replace('tmpfiles.org/', 'tmpfiles.org/dl/')
   662→   662→  } catch {
   663→   663→    clearTimeout(timeoutId)
   664→   664→    return null
   665→   665→  }

   788→   788→        const body = await res.text().catch(() => 'unknown')
   789→   789→        if (attempt < MAX_RETRIES) continue

   791→   791→      }
   792→   792→

   794→   794→      if (!ct.startsWith('image/')) {

   797→   797→      }
   798→   798→
   799→   799→      const buf = Buffer.from(await res.arrayBuffer())
   800→   800→      if (buf.length < 3000) {
   801→   801→        if (attempt < MAX_RETRIES) continue
   802→   802→        return { success: false, error: `Pollinations returned tiny image (${buf.length} bytes)` }
   803→   803→      }
   804→   804→
   805→   805→      let mime = 'image/jpeg'
   806→   806→      if (ct.includes('image/png')) mime = 'image/png'
   807→   807→      else if (ct.includes('image/webp')) mime = 'image/webp'
   808→   808→

   811→   811→      return {
   812→   812→        success: true,
   813→   813→        imageUrl: dataUrl,

   824→   824→      const msg = isTimeout ? `Pollinations timed out` : `Pollinations error: ${(err as Error).message.substring(0, 100)}`
   825→   825→      console.log(`[virtual-tryon] ${msg} (attempt ${attempt + 1})`)
   826→   826→      if (attempt < MAX_RETRIES) continue
   827→   827→      return { success: false, error: msg }

   829→   829→  }

   888→   888→
   889→   889→export async function performVirtualTryOn(input: TryOnInput): Promise<TryOnResult> {
   890→   890→  const totalStart = Date.now()
   891→   891→  const totalDeadline = totalStart + TOTAL_TIMEOUT_MS
   892→   892→  const strategiesAttempted: string[] = []
   893→   893→  const strategyErrors: Record<string, string> = {}
   894→   894→  const isVercel = !!process.env.VERCEL
   895→   895→
   896→   896→  console.log(`[virtual-tryon] v22 start: "${input.productName}" (${input.categorySlug}) — VERCEL=${isVercel}, hasSelfie=${!!input.selfieData}, hasProductImg=${!!input.productImageBase64}`)
   897→   897→
   898→   898→  // Validate selfie
   899→   899→  if (!input.selfieData?.startsWith('data:image/')) {
   900→   900→    return {
   901→   901→      success: false,
   902→   902→      error: 'A valid selfie image is required.',
   903→   903→      errorCode: 'NO_SELFIE',
   904→   904→      elapsedMs: Date.now() - totalStart,
   905→   905→      debugInfo: { strategiesAttempted, strategyErrors },
   906→   906→    }
   907→   907→  }
   908→   908→
   909→   909→  // ── STRATEGY A: Direct ZAI image-edit (LOCAL / SANDBOX only) ────
   910→   910→  // Calls ZAI's images.generations.edit with BOTH the selfie and the
   911→   911→  // product image (edit-both). This preserves the user's face/gender
   912→   912→  // AND renders the exact product. Completes in 20-27s.
   913→   913→  if (!isVercel && Date.now() < totalDeadline - 20_000) {
   914→   914→    strategiesAttempted.push('zai-image-edit')
   915→   915→    console.log('[virtual-tryon] Strategy A: Direct ZAI image-edit (edit-both)')
   916→   916→    const result = await callZAIImageEdit(input, totalDeadline)
   917→   917→    if (result.success && result.imageUrl) {
   918→   918→      const elapsed = Date.now() - totalStart
   919→   919→      console.log(`[virtual-tryon] ✅ ZAI image-edit succeeded in ${(elapsed / 1000).toFixed(1)}s`)
   920→   920→      return {
   921→   921→        success: true,
   922→   922→        imageUrl: result.imageUrl,
   923→   923→        strategy: 'zai-image-edit',
   924→   924→        elapsedMs: elapsed,
   925→   925→        debugInfo: { strategiesAttempted, strategyErrors },
   926→   926→      }
   927→   927→    }
   928→   928→    strategyErrors['zai-image-edit'] = result.error || 'No image returned'
   929→   929→    console.log(`[virtual-tryon] ZAI image-edit failed: ${result.error?.substring(0, 150)}`)
   930→   930→  } else if (isVercel) {
   931→   931→    strategiesAttempted.push('zai-image-edit-skipped')
   932→   932→    strategyErrors['zai-image-edit-skipped'] = 'Vercel environment — ZAI auth fails on public API'
   933→   933→    console.log('[virtual-tryon] Skipping ZAI (Vercel environment)')
   934→   934→  }
   935→   935→
   936→   936→  // ── STRATEGY B: Pollinations with SELFIE reference (VERCEL + FALLBACK)
   937→   937→  // Used on Vercel (always) and on local when ZAI fails. Uploads the
   938→   938→  // user's SELFIE as the Pollinations `?image=` reference so the
   939→   939→  // generated person preserves the user's face/gender/features. The
   940→   940→  // product is described in the prompt using colours EXTRACTED FROM
   941→   941→  // THE ACTUAL PRODUCT IMAGE (via sharp) — not just the product name.
   942→   942→  if (Date.now() < totalDeadline - 12_000) {
   943→   943→    strategiesAttempted.push('pollinations-selfie-img2img')
   944→   944→    console.log('[virtual-tryon] Strategy B: Pollinations (selfie reference + image-extracted colours)')
   945→   945→    const result = await callPollinationsWithSelfieReference(input, totalDeadline)
   946→   946→    if (result.success && result.imageUrl) {
   947→   947→      const elapsed = Date.now() - totalStart
   948→   948→      console.log(`[virtual-tryon] ✅ Pollinations succeeded in ${(elapsed / 1000).toFixed(1)}s`)
   949→   949→      return {
   950→   950→        success: true,
   951→   951→        imageUrl: result.imageUrl,
   952→   952→        strategy: result.strategy || 'pollinations-selfie-img2img',
   953→   953→        elapsedMs: elapsed,
   954→   954→        debugInfo: {
   955→   955→          strategiesAttempted,
   956→   956→          strategyErrors,
   957→   957→          extractedColors: result.debugInfo?.extractedColors,
   958→   958→          promptPreview: result.debugInfo?.promptPreview,
   959→   959→          selfieUploaded: result.debugInfo?.selfieUploaded,
   960→   960→        },
   961→   961→      }
   962→   962→    }
   963→   963→    strategyErrors['pollinations-selfie-img2img'] = result.error || 'No image returned'
   964→   964→    console.log(`[virtual-tryon] Pollinations failed: ${result.error?.substring(0, 150)}`)
   965→   965→  }
   966→   966→
   967→   967→  // ── All strategies failed ───────────────────────────────────────
   968→   968→  const elapsed = Date.now() - totalStart
   969→   969→  console.log(`[virtual-tryon] ❌ All strategies failed in ${(elapsed / 1000).toFixed(1)}s`)
   970→   970→  console.log(`[virtual-tryon] Strategies: ${strategiesAttempted.join(', ')}`)
   971→   971→  console.log(`[virtual-tryon] Errors: ${JSON.stringify(strategyErrors)}`)
   972→   972→
   973→   973→  return {
   974→   974→    success: false,
   975→   975→    error: 'We could not generate your style preview right now. Please try again in a moment.',
   976→   976→    errorCode: 'ALL_STRATEGIES_FAILED',
   977→   977→    elapsedMs: elapsed,
   978→   978→    debugInfo: { strategiesAttempted, strategyErrors },
   979→   979→  }
   980→   980→}
