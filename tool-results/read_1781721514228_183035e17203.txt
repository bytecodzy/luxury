     1→/**
     2→ * Virtual Try-On Engine v21 — Direct ZAI + Pollinations Fallback
     3→ *
     4→ * ─────────────────────────────────────────────────────────────────────────
     5→ *  WHY v21?
     6→ *  ─────────────────────────────────────────────────────────────────────────
     7→ *  v19 called ZAI via `ZAI.create()` (auto-discovery) from the Next.js
     8→ *  process — flaky, frequent timeouts ("Generation Timed Out").
     9→ *  v20 routed through the ai-proxy mini-service — but the ai-proxy's
    10→ *  multi-strategy sequence (2 VLM calls + 4 image strategies) triggered
    11→ *  ZAI rate-limiting, causing all strategies to fail fast.
    12→ *
    13→ *  v21 calls ZAI DIRECTLY with explicit config (read from /etc/.z-ai-config),
    14→ *  using a SINGLE `images.generations.edit` call with BOTH the selfie and
    15→ *  the product image. Direct testing confirmed this completes reliably in
    16→ *  20-27 seconds with excellent quality (preserves the user's face AND
    17→ *  renders the exact product).
    18→ *
    19→ *  ARCHITECTURE:
    20→ *    Client (browser) POST /api/try-on
    21→ *      └─► performVirtualTryOn()  (THIS FILE)
    22→ *            ├─► Strategy A: Direct ZAI image-edit (LOCAL / SANDBOX only)
    23→ *            │     • Explicit config from /etc/.z-ai-config
    24→ *            │     • edit-both: selfie + product image → preserves face + product
    25→ *            │     • 20-27s, high quality
    26→ *            └─► Strategy B: Pollinations img2img (VERCEL + fallback)
    27→ *                  • Upload product image → use as ?image= reference
    28→ *                  • Correct product always shown; model matches category gender
    29→ *                  • 10-20s, always works
    30→ *
    31→ *  ENVIRONMENT BEHAVIOUR:
    32→ *    • LOCAL / SANDBOX: Direct ZAI edit-both (best quality, preserves face
    33→ *      AND product). Falls back to Pollinations if ZAI is temporarily down.
    34→ *    • VERCEL: Pollinations img2img with product image reference (ZAI auth
    35→ *      fails on Vercel's public API). The correct product is always shown.
    36→ * ─────────────────────────────────────────────────────────────────────────
    37→ */

    93→let cachedZAIConfig: ZAIConfig | null | undefined = undefined
    94→
    95→function getZAIConfig(): ZAIConfig | null {
    96→  if (cachedZAIConfig !== undefined) return cachedZAIConfig
    97→
    98→  // On Vercel, ZAI auth fails on the public API — skip entirely
    99→  if (process.env.VERCEL) {
   100→    cachedZAIConfig = null
   101→    return null
   102→  }
   103→
   104→  // Try env vars first
   105→  if (process.env.ZAI_BASE_URL && process.env.ZAI_API_KEY) {
   106→    cachedZAIConfig = {
   107→      baseUrl: process.env.ZAI_BASE_URL,
   108→      apiKey: process.env.ZAI_API_KEY,
   109→      chatId: process.env.ZAI_CHAT_ID || '',
   110→      token: process.env.ZAI_TOKEN || '',
   111→      userId: process.env.ZAI_USER_ID || '',
   112→    }
   113→    return cachedZAIConfig
   114→  }
   115→
   116→  // Read from config files (sandbox)
   117→  const configPaths = [
   118→    '/etc/.z-ai-config',
   119→    path.join(process.cwd(), '.z-ai-config'),
   120→    path.join(os.homedir(), '.z-ai-config'),
   121→  ]
   122→  for (const filePath of configPaths) {
   123→    try {
   124→      const configStr = fs.readFileSync(filePath, 'utf-8')
   125→      const config = JSON.parse(configStr)
   126→      if (config.baseUrl && config.apiKey) {
   127→        cachedZAIConfig = {
   128→          baseUrl: config.baseUrl,
   129→          apiKey: config.apiKey,
   130→          chatId: config.chatId || '',
   131→          token: config.token || '',
   132→          userId: config.userId || '',
   133→        }
   134→        console.log(`[virtual-tryon] ZAI config loaded from ${filePath}`)
   135→        return cachedZAIConfig
   136→      }
   137→    } catch {
   138→      // continue
   139→    }
   140→  }
   141→
   142→  cachedZAIConfig = null
   143→  return null
   144→}

   148→let zaiInstanceCache: any = null

   150→async function getZAI(): Promise<any | null> {
   151→  if (process.env.VERCEL) return null
   152→  if (zaiInstanceCache) return zaiInstanceCache
   153→
   154→  const config = getZAIConfig()
   155→  if (!config) return null
   156→
   157→  try {
   158→    const ZAIModule = await import('z-ai-web-dev-sdk')
   159→    const ZAI = (ZAIModule as any).default || (ZAIModule as any)
   160→    zaiInstanceCache = new ZAI({
   161→      baseUrl: config.baseUrl,
   162→      apiKey: config.apiKey,
   163→      chatId: config.chatId,
   164→      token: config.token,
   165→      userId: config.userId,
   166→    })
   167→    console.log('[virtual-tryon] ZAI SDK instance created with explicit config')
   168→    return zaiInstanceCache
   169→  } catch (err) {
   170→    console.log('[virtual-tryon] ZAI SDK init failed:', err instanceof Error ? err.message : String(err))
   171→    return null
   172→  }
   173→}

   330→function extractColors(name: string, description?: string, tags?: string[]): string {
   331→  const text = `${name} ${description || ''} ${(tags || []).join(' ')}`.toLowerCase()
   332→  const found = new Set<string>()
   333→  for (const color of COLOR_WORDS) {
   334→    if (text.includes(color)) found.add(color)
   335→  }
   336→  return Array.from(found).slice(0, 3).join(', ')
   337→}

   361→// ── Strategy A: Direct ZAI image-edit ──────────────────────────────
   362→
   363→async function callZAIImageEdit(
   364→  input: TryOnInput,
   365→  deadline: number,
   366→): Promise<{ success: boolean; imageUrl?: string; error?: string }> {
   367→  const zai = await getZAI()
   368→  if (!zai) {
   369→    return { success: false, error: 'ZAI SDK unavailable (not configured or Vercel environment)' }
   370→  }
   371→
   372→  const config = getCategoryConfig(input.categorySlug, input.productName)
   373→  const prompt = buildEditPrompt(config, input)
   374→
   375→  // Use edit-both if we have a product image, otherwise edit-selfie
   376→  const hasProductImage = input.productImageBase64 && input.productImageBase64.startsWith('data:image/')
   377→  const images = hasProductImage
   378→    ? [{ url: input.selfieData }, { url: input.productImageBase64 }]
   379→    : [{ url: input.selfieData }]
   380→
   381→  const strategyName = hasProductImage ? 'edit-both' : 'edit-selfie'
   382→  const remaining = Math.min(ZAI_EDIT_TIMEOUT_MS, deadline - Date.now() - 3_000)
   383→  if (remaining < 15_000) {
   384→    return { success: false, error: `insufficient time budget (${remaining}ms) for ZAI edit` }
   385→  }
   386→
   387→  console.log(`[virtual-tryon] ZAI image-edit (${strategyName}): ${config.size}, timeout=${remaining}ms`)
   388→  console.log(`[virtual-tryon] Prompt (first 200): ${prompt.substring(0, 200)}...`)
   389→
   390→  try {
   391→    const start = Date.now()
   392→    const response = await Promise.race([
   393→      zai.images.generations.edit({
   394→        prompt,
   395→        images,
   396→        size: config.size,
   397→      }),
   398→      new Promise<never>((_, rej) => setTimeout(() => rej(new Error('ZAI edit timeout')), remaining)),
   399→    ])
   400→    const elapsed = ((Date.now() - start) / 1000).toFixed(1)
   401→
   402→    const item = response?.data?.[0]
   403→    const b64 = item?.base64
   404→    if (!b64 || typeof b64 !== 'string' || b64.length < 3000) {
   405→      return { success: false, error: `ZAI edit returned no usable image after ${elapsed}s (base64 length=${b64?.length || 0})` }
   406→    }
   407→
   408→    // Detect format
   409→    let mime = 'image/png'
   410→    if (item.format === 'jpeg' || item.format === 'jpg') mime = 'image/jpeg'
   411→    else if (item.format === 'webp') mime = 'image/webp'
   412→    else {
   413→      const head = Buffer.from(b64.substring(0, 8), 'base64').toString('hex')
   414→      if (head.startsWith('ffd8ff')) mime = 'image/jpeg'
   415→      else if (head.startsWith('89504e47')) mime = 'image/png'
   416→      else if (head.startsWith('52494646')) mime = 'image/webp'
   417→    }
   418→
   419→    const dataUrl = `data:${mime};base64,${b64}`
   420→    const byteLen = Math.floor(b64.length * 0.75)
   421→    console.log(`[virtual-tryon] ✅ ZAI ${strategyName} succeeded in ${elapsed}s (${(byteLen / 1024).toFixed(1)}KB, ${mime})`)
   422→    return { success: true, imageUrl: dataUrl }
   423→  } catch (err) {
   424→    const msg = err instanceof Error ? err.message : String(err)
   425→    console.log(`[virtual-tryon] ZAI ${strategyName} failed: ${msg.substring(0, 200)}`)
   426→    return { success: false, error: msg.substring(0, 200) }
   427→  }
   428→}

   437→async function uploadToTmpfiles(buf: Buffer, timeoutMs: number): Promise<string | null> {
   438→  const boundary = '----3boxesTryon' + Math.random().toString(16).slice(2)
   439→  const header = Buffer.from(
   440→    `--${boundary}\r\nContent-Disposition: form-data; name="file"; filename="image.jpg"\r\nContent-Type: image/jpeg\r\n\r\n`
   441→  )
   442→  const footer = Buffer.from(`\r\n--${boundary}--\r\n`)
   443→  const body = Buffer.concat([header, buf, footer])
   444→
   445→  const controller = new AbortController()
   446→  const timeoutId = setTimeout(() => controller.abort(), timeoutMs)
   447→  try {
   448→    const res = await fetch('https://tmpfiles.org/api/v1/upload', {
   449→      method: 'POST',
   450→      headers: {
   451→        'Content-Type': `multipart/form-data; boundary=${boundary}`,
   452→        'User-Agent': '3BOXES-TryOn/1.0',
   453→      },
   454→      body,
   455→      signal: controller.signal,
   456→    })
   457→    clearTimeout(timeoutId)
   458→    if (!res.ok) return null
   459→    const json = (await res.json()) as { data?: { url?: string } }
   460→    const viewerUrl = json?.data?.url
   461→    if (!viewerUrl) return null
   462→    return viewerUrl.replace('tmpfiles.org/', 'tmpfiles.org/dl/')
   463→  } catch {
   464→    clearTimeout(timeoutId)
   465→    return null
   466→  }
   467→}

   469→function buildPollinationsPrompt(config: CategoryConfig, input: TryOnInput): string {

   471→  const genderWord = config.gender === 'woman' ? 'woman' : config.gender === 'man' ? 'man' : config.gender === 'child' ? 'child' : 'person'
   472→
   473→  const parts: string[] = []
   474→  parts.push(`Professional fashion photograph of a ${genderWord} ${config.placement}.`)
   475→  parts.push(`The product is "${input.productName}".`)
   476→  if (colors) parts.push(`The product colours are ${colors}.`)
   477→  if (config.materialHint) parts.push(`Made of ${config.materialHint}.`)
   478→  if (input.productDescription) {
   479→    const desc = input.productDescription.substring(0, 200).replace(/\s+/g, ' ').trim()
   480→    if (desc) parts.push(`Product details: ${desc}.`)
   481→  }
   482→  if (input.productTags && input.productTags.length > 0) {
   483→    const tags = input.productTags.filter(t => !['new-arrival', 'featured', 'bestseller'].includes(t)).slice(0, 5)
   484→    if (tags.length > 0) parts.push(`Style: ${tags.join(', ')}.`)
   485→  }
   486→  parts.push(`${config.framing}, studio lighting, photorealistic, sharp focus, high detail.`)
   487→  parts.push(`The product must look naturally worn with realistic folds and fit.`)
   488→  parts.push(`No glasses, no sunglasses, no hats, no extra props — only the product described.`)
   489→  return parts.join(' ')
   490→}

   492→async function callPollinationsWithProductImage(
   493→  input: TryOnInput,
   494→  deadline: number,
   495→): Promise<{ success: boolean; imageUrl?: string; error?: string; strategy?: string }> {
   496→  const config = getCategoryConfig(input.categorySlug, input.productName)
   497→  const prompt = buildPollinationsPrompt(config, input)
   498→  const { width, height } = parseImageSize(config.size)
   499→
   500→  console.log(`[virtual-tryon] Pollinations: gender=${config.gender}, ${width}x${height}`)
   501→  console.log(`[virtual-tryon] Prompt (first 200): ${prompt.substring(0, 200)}...`)
   502→
   503→  // Upload the product image so Pollinations can use it as a reference
   504→  let productUrl: string | null = null
   505→  if (input.productImageBase64 && Date.now() < deadline - 20_000) {
   506→    try {
   507→      const raw = stripDataUrl(input.productImageBase64)
   508→      const buf = Buffer.from(raw, 'base64')
   509→      productUrl = await uploadToTmpfiles(buf, UPLOAD_TIMEOUT_MS)
   510→      console.log(`[virtual-tryon] Product image uploaded: ${productUrl ? 'yes' : 'no'}`)
   511→    } catch (err) {
   512→      console.log(`[virtual-tryon] Product upload failed: ${err instanceof Error ? err.message : String(err)}`)
   513→    }
   514→  }
   515→
   516→  const encoded = encodeURIComponent(prompt)
   517→  const seed = Math.floor(Math.random() * 1_000_000)
   518→  let url = `https://image.pollinations.ai/prompt/${encoded}?width=${width}&height=${height}&model=flux&nologo=true&seed=${seed}`
   519→  if (productUrl) {
   520→    url += `&image=${encodeURIComponent(productUrl)}`
   521→  }
   522→
   523→  // Retry up to 2 times on rate-limit (HTTP 402)
   524→  const MAX_RETRIES = 2
   525→  const RETRY_DELAYS_MS = [4_000, 6_000]
   526→
   527→  for (let attempt = 0; attempt <= MAX_RETRIES; attempt++) {
   528→    if (attempt > 0) {
   529→      const delay = RETRY_DELAYS_MS[Math.min(attempt - 1, RETRY_DELAYS_MS.length - 1)]
   530→      console.log(`[virtual-tryon] Pollinations retry ${attempt}/${MAX_RETRIES} after ${delay}ms...`)
   531→      await new Promise(r => setTimeout(r, delay))
   532→    }
   533→
   534→    const retrySeed = seed + attempt * 11111
   535→    const attemptUrl = url.replace(/&seed=\d+/, `&seed=${retrySeed}`)
   536→    const remaining = Math.min(POLLINATIONS_TIMEOUT_MS, deadline - Date.now() - 3_000)
   537→    if (remaining < 8_000) {
   538→      return { success: false, error: `insufficient time for Pollinations (${remaining}ms)` }
   539→    }
   540→
   541→    const controller = new AbortController()
   542→    const timeoutId = setTimeout(() => controller.abort(), remaining)
   543→
   544→    try {
   545→      const start = Date.now()
   546→      const res = await fetch(attemptUrl, {
   547→        signal: controller.signal,
   548→        headers: { 'Accept': 'image/jpeg, image/png, image/webp, */*', 'User-Agent': '3BOXES-VirtualTryOn/1.0' },
   549→      })
   550→      clearTimeout(timeoutId)
   551→      const elapsed = ((Date.now() - start) / 1000).toFixed(1)
   552→
   553→      if (res.status === 402 && attempt < MAX_RETRIES) {
   554→        console.log(`[virtual-tryon] Pollinations 402 rate-limited after ${elapsed}s — will retry`)
   555→        continue
   556→      }
   557→      if (!res.ok) {
   558→        const body = await res.text().catch(() => 'unknown')
   559→        if (attempt < MAX_RETRIES) continue
   560→        return { success: false, error: `Pollinations HTTP ${res.status}: ${body.substring(0, 100)}` }
   561→      }
   562→
   563→      const ct = res.headers.get('content-type') || ''
   564→      if (!ct.startsWith('image/')) {
   565→        if (attempt < MAX_RETRIES) continue
   566→        return { success: false, error: `Pollinations non-image response: ${ct}` }
   567→      }
   568→
   569→      const buf = Buffer.from(await res.arrayBuffer())
   570→      if (buf.length < 3000) {
   571→        if (attempt < MAX_RETRIES) continue
   572→        return { success: false, error: `Pollinations returned tiny image (${buf.length} bytes)` }
   573→      }
   574→
   575→      let mime = 'image/jpeg'
   576→      if (ct.includes('image/png')) mime = 'image/png'
   577→      else if (ct.includes('image/webp')) mime = 'image/webp'
   578→
   579→      const dataUrl = `data:${mime};base64,${buf.toString('base64')}`
   580→      console.log(`[virtual-tryon] ✅ Pollinations succeeded in ${elapsed}s (${(buf.length / 1024).toFixed(1)}KB)`)
   581→      return { success: true, imageUrl: dataUrl, strategy: productUrl ? 'pollinations-product-img2img' : 'pollinations-text' }
   582→    } catch (err) {
   583→      clearTimeout(timeoutId)
   584→      const isTimeout = err instanceof DOMException && err.name === 'AbortError'
   585→      const msg = isTimeout ? `Pollinations timed out` : `Pollinations error: ${(err as Error).message.substring(0, 100)}`
   586→      console.log(`[virtual-tryon] ${msg} (attempt ${attempt + 1})`)
   587→      if (attempt < MAX_RETRIES) continue
   588→      return { success: false, error: msg }
   589→    }
   590→  }
   591→
   592→  return { success: false, error: 'Pollinations failed after all retries' }
   593→}

   650→export async function performVirtualTryOn(input: TryOnInput): Promise<TryOnResult> {
   651→  const totalStart = Date.now()
   652→  const totalDeadline = totalStart + TOTAL_TIMEOUT_MS
   653→  const strategiesAttempted: string[] = []
   654→  const strategyErrors: Record<string, string> = {}
   655→  const isVercel = !!process.env.VERCEL
   656→
   657→  console.log(`[virtual-tryon] v21 start: "${input.productName}" (${input.categorySlug}) — VERCEL=${isVercel}, hasSelfie=${!!input.selfieData}, hasProductImg=${!!input.productImageBase64}`)
   658→
   659→  // Validate selfie
   660→  if (!input.selfieData?.startsWith('data:image/')) {
   661→    return {
   662→      success: false,
   663→      error: 'A valid selfie image is required.',
   664→      errorCode: 'NO_SELFIE',
   665→      elapsedMs: Date.now() - totalStart,
   666→      debugInfo: { strategiesAttempted, strategyErrors },
   667→    }
   668→  }
   669→
   670→  // ── STRATEGY A: Direct ZAI image-edit (LOCAL / SANDBOX only) ────
   671→  // Calls ZAI's images.generations.edit with BOTH the selfie and the
   672→  // product image (edit-both). This preserves the user's face/gender
   673→  // AND renders the exact product. Completes in 20-27s.
   674→  if (!isVercel && Date.now() < totalDeadline - 20_000) {
   675→    strategiesAttempted.push('zai-image-edit')
   676→    console.log('[virtual-tryon] Strategy A: Direct ZAI image-edit (edit-both)')
   677→    const result = await callZAIImageEdit(input, totalDeadline)
   678→    if (result.success && result.imageUrl) {
   679→      const elapsed = Date.now() - totalStart
   680→      console.log(`[virtual-tryon] ✅ ZAI image-edit succeeded in ${(elapsed / 1000).toFixed(1)}s`)
   681→      return {
   682→        success: true,
   683→        imageUrl: result.imageUrl,
   684→        strategy: 'zai-image-edit',
   685→        elapsedMs: elapsed,
   686→        debugInfo: { strategiesAttempted, strategyErrors },
   687→      }
   688→    }
   689→    strategyErrors['zai-image-edit'] = result.error || 'No image returned'
   690→    console.log(`[virtual-tryon] ZAI image-edit failed: ${result.error?.substring(0, 150)}`)
   691→  } else if (isVercel) {
   692→    strategiesAttempted.push('zai-image-edit-skipped')
   693→    strategyErrors['zai-image-edit-skipped'] = 'Vercel environment — ZAI auth fails on public API'
   694→    console.log('[virtual-tryon] Skipping ZAI (Vercel environment)')
   695→  }
   696→
   697→  // ── STRATEGY B: Pollinations with product image (FALLBACK) ──────
   698→  // Used on Vercel (always) and on local when ZAI fails. Uses the
   699→  // PRODUCT IMAGE as the Pollinations `?image=` reference so the
   700→  // generated image shows the CORRECT product. The person is a model
   701→  // matching the category's gender.
   702→  if (Date.now() < totalDeadline - 12_000) {
   703→    strategiesAttempted.push('pollinations-product-img2img')
   704→    console.log('[virtual-tryon] Strategy B: Pollinations (product image reference)')
   705→    const result = await callPollinationsWithProductImage(input, totalDeadline)
   706→    if (result.success && result.imageUrl) {
   707→      const elapsed = Date.now() - totalStart
   708→      console.log(`[virtual-tryon] ✅ Pollinations succeeded in ${(elapsed / 1000).toFixed(1)}s`)
   709→      return {
   710→        success: true,
   711→        imageUrl: result.imageUrl,
   712→        strategy: result.strategy || 'pollinations-product-img2img',
   713→        elapsedMs: elapsed,
   714→        debugInfo: { strategiesAttempted, strategyErrors },
   715→      }
   716→    }
   717→    strategyErrors['pollinations-product-img2img'] = result.error || 'No image returned'
   718→    console.log(`[virtual-tryon] Pollinations failed: ${result.error?.substring(0, 150)}`)
   719→  }
   720→
   721→  // ── All strategies failed ───────────────────────────────────────
   722→  const elapsed = Date.now() - totalStart
   723→  console.log(`[virtual-tryon] ❌ All strategies failed in ${(elapsed / 1000).toFixed(1)}s`)
   724→  console.log(`[virtual-tryon] Strategies: ${strategiesAttempted.join(', ')}`)
   725→  console.log(`[virtual-tryon] Errors: ${JSON.stringify(strategyErrors)}`)
   726→
   727→  return {
   728→    success: false,
   729→    error: 'We could not generate your style preview right now. Please try again in a moment.',
   730→    errorCode: 'ALL_STRATEGIES_FAILED',
   731→    elapsedMs: elapsed,
   732→    debugInfo: { strategiesAttempted, strategyErrors },
   733→  }
   734→}
