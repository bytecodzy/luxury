     1→/**
     2→ * Virtual Try-On Engine v19 — ZAI IMAGE-EDIT Pipeline (Selfie-Preserving)
     3→ *
     4→ * ─────────────────────────────────────────────────────────────────────────
     5→ *  WHY v19?
     6→ *  ─────────────────────────────────────────────────────────────────────────
     7→ *  v18 used Pollinations.ai with the selfie URL passed via `?image=`. In
     8→ *  practice Pollinations does NOT honour that parameter as a true
     9→ *  image-to-image conditioning signal — it falls back to plain text-to-image
    10→ *  and produces random people wearing random products. That is exactly what
    11→ *  the user saw: "dress or saree → showing glasses, selfie not loading".
    12→ *
    13→ *  v19 uses the Z.AI `images.generations.edit` endpoint, which is a REAL
    14→ *  image-to-image edit model:
    15→ *    • INPUT  = the user's SELFIE (compressed)
    16→ *    • PROMPT = a rich description of the PRODUCT to drape on the person
    17→ *    • OUTPUT = the SAME person (face, gender, skin tone, body, hair) now
    18→ *               wearing/holding the described product, photorealistically
    19→ *               blended with realistic shadows, folds and lighting.
    20→ *
    21→ *  We confirmed in the sandbox that:
    22→ *    • `zai.images.generations.create({...})` works (≈28s)
    23→ *    • `zai.images.generations.edit({prompt, images:[{url}], size})` works
    24→ *      (≈14–18s) and returns a 75–110KB edited PNG.
    25→ *
    26→ *  We ALSO use ZAI VLM (`chat.completions.createVision`) to extract a rich
    27→ *  description of the ACTUAL product photo (colours, pattern, material,
    28→ *  style, type). This is fused with the product name/description/tags so
    29→ *  the edit prompt accurately reflects the EXACT product being tried on —
    30→ *  no more "selected saree → got glasses".
    31→ *
    32→ * ─────────────────────────────────────────────────────────────────────────
    33→ *  STRATEGY CHAIN
    34→ *  ─────────────────────────────────────────────────────────────────────────
    35→ *  1. ZAI image-edit (PRIMARY)         — best quality, preserves user.
    36→ *  2. ZAI text-to-image (FALLBACK 1)   — if edit fails but SDK is alive.
    37→ *  3. Pollinations img2img (FALLBACK 2) — if ZAI is unavailable, attempt
    38→ *     to upload the selfie to tmpfiles.org and use it as `?image=`.
    39→ *  4. Pollinations text-to-image (LAST RESORT) — always available.
    40→ *
    41→ *  All paths return the same `{success, imageUrl, strategy}` contract.
    42→ *
    43→ * ─────────────────────────────────────────────────────────────────────────
    44→ *  ENVIRONMENT
    45→ *  ─────────────────────────────────────────────────────────────────────────
    46→ *  • Sandbox / preview: ZAI.create() auto-discovers credentials and works.
    47→ *  • Vercel: requires `ZAI_BASE_URL` and `ZAI_API_KEY` env vars. If they
    48→ *    are missing, we transparently fall back to Pollinations so the user
    49→ *    still gets a result (no "AI is busy" error).
    50→ */
    51→
    52→// NOTE: `sharp` is imported LAZILY via getSharp() below — NOT at the top
    53→// level. This is critical for Vercel: a top-level `import sharp from 'sharp'`
    54→// would try to load the native binary when the module is first imported (e.g.
    55→// by the /api/try-on/status route), and if the binary is missing/incompatible
    56→// in the serverless environment it crashes the entire function with HTTP 500.
    57→// Lazy-loading confines any sharp failure to the actual try-on request.
    58→
    59→let sharpModule: typeof import('sharp') | null = null
    60→async function getSharp() {
    61→  if (sharpModule) return sharpModule
    62→  const mod = await import('sharp')
    63→  sharpModule = (mod as any).default || mod
    64→  return sharpModule
    65→}

   267→function getCategoryConfig(categorySlug: string, productName: string): CategoryTryOnConfig {
   268→  if (CATEGORY_CONFIG[categorySlug]) {
   269→    const config = { ...CATEGORY_CONFIG[categorySlug] }
   270→    // Refine jewelry placement based on product name
   271→    if (categorySlug.includes('jewel')) {
   272→      const n = productName.toLowerCase()
   273→      if (n.includes('earring') || n.includes('jhumka') || n.includes('stud'))
   274→        config.placement = 'wearing the earrings on both earlobes, the earrings clearly visible and properly positioned'
   275→      else if (n.includes('necklace') || n.includes('choker') || n.includes('pendant') || n.includes('temple') || n.includes('haar') || n.includes('mala'))
   276→        config.placement = 'wearing the necklace around the neck, the chain sitting naturally at the collarbone'
   277→      else if (n.includes('bracelet') || n.includes('cuff') || n.includes('bangle') || n.includes('kada'))
   278→        config.placement = 'wearing the bracelet on the wrist, properly fitted'
   279→      else if (n.includes('ring'))
   280→        config.placement = 'wearing the ring on the finger, the ring clearly visible'
   281→      else if (n.includes('set') || n.includes('bridal'))
   282→        config.placement = 'wearing a matching jewelry set — necklace around the neck and earrings on both earlobes'
   283→    }
   284→    return config
   285→  }
   286→
   287→  // Fuzzy match
   288→  const knownSlugs = Object.keys(CATEGORY_CONFIG)
   289→  const matched = knownSlugs.find(s => categorySlug.includes(s) || s.includes(categorySlug))
   290→  if (matched) return { ...CATEGORY_CONFIG[matched] }

   292→  return {

   296→    framing: 'upper-body to three-quarter photograph',

   317→function extractColors(name: string, description?: string, tags?: string[]): string {
   318→  const text = `${name} ${description || ''} ${(tags || []).join(' ')}`.toLowerCase()
   319→  const found = new Set<string>()
   320→  for (const color of COLOR_WORDS) {
   321→    if (text.includes(color)) found.add(color)
   322→  }
   323→  return Array.from(found).slice(0, 3).join(', ')
   324→}

   326→/**
   327→ * Build a rich, detailed description of the PRODUCT for the prompt.
   328→ * This is what tells the AI WHAT to drape on the person.
   329→ */
   330→function buildProductDescription(
   331→  config: CategoryTryOnConfig,
   332→  input: TryOnInput,
   333→  vlmDescription?: string,
   334→): string {
   335→  const colors = extractColors(input.productName, input.productDescription, input.productTags)
   336→  const parts: string[] = []
   337→
   338→  parts.push(`a ${config.garmentType}`)
   339→
   340→  if (colors) {
   341→    parts.push(`in ${colors}`)
   342→  }
   343→
   344→  // Add the product name for specificity
   345→  parts.push(`specifically "${input.productName}"`)
   346→
   347→  // Add material hint
   348→  if (config.materialHint) {
   349→    parts.push(`made of ${config.materialHint}`)
   350→  }
   351→
   352→  // If VLM gave us a description of the actual product photo, USE IT —
   353→  // this is the single most important signal for matching the product.
   354→  if (vlmDescription && vlmDescription.trim().length > 10) {
   355→    parts.push(`(visual reference: ${vlmDescription.trim().substring(0, 280)})`)
   356→  } else if (input.productDescription) {
   357→    // Fallback to product description text (first 160 chars)
   358→    const desc = input.productDescription.substring(0, 160).replace(/\s+/g, ' ').trim()
   359→    if (desc) {
   360→      parts.push(`(product details: ${desc})`)
   361→    }
   362→  }
   363→
   364→  // Add tags if available
   365→  if (input.productTags && input.productTags.length > 0) {
   366→    const relevantTags = input.productTags
   367→      .filter(t => !['new-arrival', 'featured', 'bestseller'].includes(t))
   368→      .slice(0, 5)
   369→    if (relevantTags.length > 0) {
   370→      parts.push(`with ${relevantTags.join(', ')} characteristics`)
   371→    }
   372→  }
   373→
   374→  return parts.join(' ')
   375→}

   379→function stripDataUrl(dataUrl: string): string {
   380→  const match = dataUrl.match(/^data:image\/[^;]+;base64,(.+)$/)
   381→  return match ? match[1] : dataUrl
   382→}

   384→/**
   385→ * Compress a selfie data URL into a thumbnail suitable for ZAI image-edit.
   386→ * Keeps the longest edge at 1024px (enough for the AI to preserve face/body,
   387→ * small enough for fast upload + processing).
   388→ */
   389→async function compressSelfie(selfieDataUrl: string): Promise<Buffer> {
   390→  const sharp = await getSharp()
   391→  const raw = stripDataUrl(selfieDataUrl)
   392→  const inputBuf = Buffer.from(raw, 'base64')
   393→  return sharp(inputBuf)
   394→    .resize(1024, 1280, { fit: 'inside', withoutEnlargement: true })
   395→    .jpeg({ quality: 85, mozjpeg: true })
   396→    .toBuffer()
   397→}
   398→
   399→/**
   400→ * Convert a compressed selfie buffer back to a data URL (ZAI edit needs a URL).
   401→ */
   402→function bufferToDataUrl(buf: Buffer, mime = 'image/jpeg'): string {
   403→  return `data:${mime};base64,${buf.toString('base64')}`
   404→}

   406→/**
   407→ * Compress a product image for VLM analysis — small is fine, the VLM only
   408→ * needs to identify colours, type, pattern, material.
   409→ */
   410→async function compressProductImageForVLM(productImageBase64: string): Promise<string | null> {
   411→  if (!productImageBase64) return null
   412→  try {
   413→    const sharp = await getSharp()
   414→    const raw = stripDataUrl(productImageBase64)
   415→    const inputBuf = Buffer.from(raw, 'base64')
   416→    const out = await sharp(inputBuf)
   417→      .resize(768, 768, { fit: 'inside', withoutEnlargement: true })
   418→      .jpeg({ quality: 75, mozjpeg: true })
   419→      .toBuffer()
   420→    return bufferToDataUrl(out)
   421→  } catch {
   422→    return null
   423→  }
   424→}

   426→/**
   427→ * Upload a compressed selfie buffer to tmpfiles.org (free, anonymous, no auth).
   428→ * Used by the Pollinations fallback path.
   429→ */
   430→async function uploadToTmpfiles(buf: Buffer, timeoutMs: number): Promise<string | null> {
   431→  const boundary = '----3boxesTryon' + Math.random().toString(16).slice(2)
   432→  const filename = 'selfie.jpg'
   433→  const header = Buffer.from(
   434→    `--${boundary}\r\nContent-Disposition: form-data; name="file"; filename="${filename}"\r\nContent-Type: image/jpeg\r\n\r\n`
   435→  )
   436→  const footer = Buffer.from(`\r\n--${boundary}--\r\n`)
   437→  const body = Buffer.concat([header, buf, footer])
   438→
   439→  const controller = new AbortController()
   440→  const timeoutId = setTimeout(() => controller.abort(), timeoutMs)
   441→
   442→  try {
   443→    const start = Date.now()
   444→    const res = await fetch('https://tmpfiles.org/api/v1/upload', {
   445→      method: 'POST',
   446→      headers: {
   447→        'Content-Type': `multipart/form-data; boundary=${boundary}`,
   448→        'User-Agent': '3BOXES-TryOn/1.0',
   449→      },
   450→      body,
   451→      signal: controller.signal,
   452→    })
   453→    clearTimeout(timeoutId)
   454→    const elapsed = ((Date.now() - start) / 1000).toFixed(1)
   455→
   456→    if (!res.ok) {
   457→      console.log(`[virtual-tryon] tmpfiles upload HTTP ${res.status} after ${elapsed}s`)
   458→      return null
   459→    }
   460→
   461→    const json = (await res.json()) as { status?: string; data?: { url?: string } }
   462→    const viewerUrl = json?.data?.url
   463→    if (!viewerUrl || typeof viewerUrl !== 'string') {
   464→      console.log(`[virtual-tryon] tmpfiles returned no url after ${elapsed}s`)
   465→      return null
   466→    }
   467→
   468→    const directUrl = viewerUrl.replace('tmpfiles.org/', 'tmpfiles.org/dl/')
   469→    console.log(`[virtual-tryon] tmpfiles uploaded in ${elapsed}s → ${directUrl.substring(0, 60)}...`)
   470→    return directUrl
   471→  } catch (err) {
   472→    clearTimeout(timeoutId)
   473→    const isTimeout = err instanceof DOMException && err.name === 'AbortError'
   474→    console.log(`[virtual-tryon] tmpfiles upload failed: ${isTimeout ? 'timeout' : (err as Error).message}`)
   475→    return null
   476→  }
   477→}

   484→async function getZAI(): Promise<any | null> {
   485→  // ── VERCEL FAST-PATH ────────────────────────────────────────────
   486→  // On Vercel, the ZAI SDK cannot authenticate:
   487→  //   • The public endpoint `api.z.ai/api/v1` REJECTS the `"Z.ai"` apiKey
   488→  //     with "Authentication Failed" (verified by direct curl tests).
   489→  //   • The `images.generations.edit` endpoint returns 404 NOT_FOUND on the
   490→  //     public API (it only exists on `internal-api.z.ai`, which is a private
   491→  //     network inaccessible from Vercel).
   492→  //   • The JWT token from the sandbox config is also rejected ("token
   493→  //     expired or incorrect") on the public API.
   494→  // Trying to init the SDK on Vercel wastes 15-30s on dead/hung connections
   495→  // before failing, which pushes the total past Vercel's 60s function limit

   763→async function callPollinations(
   764→  prompt: string,
   765→  size: ImageSize,
   766→  timeoutMs: number,
   767→  referenceImageUrl?: string,
   768→): Promise<{ success: boolean; imageUrl?: string; error?: string }> {
   769→  const { width, height } = parseImageSize(size)
   770→  const mode = referenceImageUrl ? 'selfie-img2img' : 'text'
   771→
   772→  // Pollinations sometimes returns HTTP 402 "Queue full for IP" when the
   773→  // caller's IP already has a request in flight. We retry up to 2 times with
   774→  // a short backoff so transient rate-limiting doesn't fail the try-on.
   775→  const MAX_RETRIES = 2
   776→  const RETRY_DELAYS_MS = [4_000, 6_000] // wait 4s, then 6s before retrying
   777→
   778→  for (let attempt = 0; attempt <= MAX_RETRIES; attempt++) {
   779→    if (attempt > 0) {
   780→      const delay = RETRY_DELAYS_MS[Math.min(attempt - 1, RETRY_DELAYS_MS.length - 1)]
   781→      console.log(`[virtual-tryon] Pollinations retry ${attempt}/${MAX_RETRIES} after ${delay}ms backoff...`)
   782→      await new Promise(r => setTimeout(r, delay))
   783→    }
   784→
   785→    const encoded = encodeURIComponent(prompt)
   786→    const seed = Math.floor(Math.random() * 1_000_000)
   787→    let url = `https://image.pollinations.ai/prompt/${encoded}?width=${width}&height=${height}&model=flux&nologo=true&seed=${seed}`
   788→    if (referenceImageUrl) {
   789→      url += `&image=${encodeURIComponent(referenceImageUrl)}`
   790→    }
   791→
   792→    console.log(`[virtual-tryon] Pollinations ${mode}: ${width}x${height} (attempt ${attempt + 1}/${MAX_RETRIES + 1}, timeout ${timeoutMs}ms)`)
   793→
   794→    const controller = new AbortController()
   795→    const timeoutId = setTimeout(() => controller.abort(), timeoutMs)
   796→
   797→    try {
   798→      const start = Date.now()
   799→      const res = await fetch(url, {
   800→        signal: controller.signal,
   801→        headers: {
   802→          'Accept': 'image/jpeg, image/png, image/webp, */*',
   803→          'User-Agent': '3BOXES-VirtualTryOn/1.0',
   804→        },
   805→      })
   806→      clearTimeout(timeoutId)
   807→      const elapsed = ((Date.now() - start) / 1000).toFixed(1)
   808→
   809→      // ── Rate-limited (HTTP 402) — retry with backoff ──
   810→      if (res.status === 402 && attempt < MAX_RETRIES) {
   811→        const body = await res.text().catch(() => '')
   812→        console.log(`[virtual-tryon] Pollinations 402 rate-limited after ${elapsed}s (attempt ${attempt + 1}) — will retry`)
   813→        continue // retry
   814→      }
   815→
   816→      if (!res.ok) {
   817→        const body = await res.text().catch(() => 'unknown')
   818→        return { success: false, error: `Pollinations HTTP ${res.status} after ${elapsed}s: ${body.substring(0, 150)}` }
   819→      }
   820→
   821→      const ct = res.headers.get('content-type') || ''
   822→      if (!ct.startsWith('image/')) {
   823→        const body = await res.text().catch(() => 'unknown')
   824→        return { success: false, error: `Pollinations non-image "${ct}" after ${elapsed}s: ${body.substring(0, 150)}` }
   825→      }
   826→
   827→      const buf = Buffer.from(await res.arrayBuffer())
   828→      if (buf.length < 3000) {
   829→        return { success: false, error: `Pollinations returned tiny image (${buf.length} bytes) — likely an error` }
   830→      }
   831→
   832→      let mime = 'image/jpeg'
   833→      if (ct.includes('image/png')) mime = 'image/png'
   834→      else if (ct.includes('image/webp')) mime = 'image/webp'
   835→      else {
   836→        const hex = buf.subarray(0, 4).toString('hex')
   837→        if (hex === '89504e47') mime = 'image/png'
   838→        else if (hex.startsWith('ffd8ff')) mime = 'image/jpeg'
   839→        else if (hex.startsWith('52494646')) mime = 'image/webp'
   840→      }
   841→
   842→      const dataUrl = `data:${mime};base64,${buf.toString('base64')}`
   843→      console.log(`[virtual-tryon] ✅ Pollinations ${mode} succeeded in ${elapsed}s (${(buf.length / 1024).toFixed(1)}KB, ${mime})`)
   844→      return { success: true, imageUrl: dataUrl }
   845→    } catch (err) {
   846→      clearTimeout(timeoutId)
   847→      const isTimeout = err instanceof DOMException && err.name === 'AbortError'
   848→      const msg = isTimeout
   849→        ? `Pollinations timed out after ${(timeoutMs / 1000).toFixed(0)}s`
   850→        : `Pollinations fetch error: ${(err as Error).message.substring(0, 200)}`
   851→      console.log(`[virtual-tryon] ${msg} (attempt ${attempt + 1})`)
   852→      if (attempt < MAX_RETRIES) continue // retry
   853→      return { success: false, error: msg }
   854→    }
   855→  }
   856→
   857→  return { success: false, error: 'Pollinations failed after all retries' }
   858→}

   959→    // back to the raw decoded selfie buffer. This is larger but still works
