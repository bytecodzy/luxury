    76→function compressImage(dataUrl: string, maxSize = 1024, quality = 0.85): Promise<string> {
    77→  return new Promise((resolve, reject) => {
    78→    const img = document.createElement('img');
    79→    img.onload = () => {
    80→      const canvas = document.createElement('canvas');
    81→      let { width, height } = img;
    82→
    83→      if (width > maxSize || height > maxSize) {
    84→        if (width > height) {
    85→          height = Math.round((height * maxSize) / width);
    86→          width = maxSize;
    87→        } else {
    88→          width = Math.round((width * maxSize) / height);
    89→          height = maxSize;
    90→        }
    91→      }
    92→
    93→      canvas.width = width;
    94→      canvas.height = height;
    95→      const ctx = canvas.getContext('2d');
    96→      if (!ctx) {
    97→        reject(new Error('Could not get canvas context'));
    98→        return;
    99→      }
   100→      ctx.drawImage(img, 0, 0, width, height);
   101→      resolve(canvas.toDataURL('image/jpeg', quality));
   102→    };
   103→    img.onerror = () => reject(new Error('Failed to load image'));
   104→    img.src = dataUrl;
   105→  });
   106→}

   108→// ── Helper: Fetch image as base64 ──────────────────────────────────

   110→async function fetchImageAsBase64(url: string): Promise<string | null> {
   111→  if (url.startsWith('data:')) return url;
   112→
   113→  const strategies: string[] = [];
   114→
   115→  if (url.startsWith('//')) {
   116→    strategies.push(`/api/image-proxy?url=${encodeURIComponent(`https:${url}`)}`);
   117→  } else if (url.startsWith('http://') || url.startsWith('https://')) {
   118→    strategies.push(`/api/image-proxy?url=${encodeURIComponent(url)}`);
   119→    strategies.push(url);
   120→  } else if (url.startsWith('/') && !url.startsWith('/api/')) {
   121→    strategies.push(url);
   122→  }
   123→
   124→  for (const fetchUrl of strategies) {
   125→    try {
   126→      const response = await fetch(fetchUrl, { signal: AbortSignal.timeout(8000) });
   127→      if (!response.ok) continue;
   128→      const contentType = response.headers.get('content-type') || '';
   129→      if (!contentType.startsWith('image/') && !contentType.startsWith('application/octet-stream')) continue;
   130→      const blob = await response.blob();
   131→      const base64 = await new Promise<string>((resolve) => {
   132→        const reader = new FileReader();
   133→        reader.onloadend = () => resolve(reader.result as string);
   134→        reader.readAsDataURL(blob);
   135→      });
   136→      if (base64.startsWith('data:')) return base64;
   137→    } catch {}
   138→  }
   139→  return null;
   140→}

   145→function add3BoxesWatermark(imageDataUrl: string, productName: string): Promise<string> {
   146→  return new Promise((resolve) => {
   147→    try {
   148→      const img = document.createElement('img');
   149→      img.crossOrigin = 'anonymous';

   151→        try {
   152→          const canvas = document.createElement('canvas');
   153→          // Use the FULL image dimensions — never crop
   154→          canvas.width = img.naturalWidth;
   155→          canvas.height = img.naturalHeight;
   156→          const ctx = canvas.getContext('2d');
   157→          if (!ctx) {
   158→            resolve(imageDataUrl);
   159→            return;
   160→          }
   161→
   162→          // Draw the original image at FULL size
   163→          ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
   164→
   165→          const w = canvas.width;
   166→          const h = canvas.height;
   167→
   168→          // ── Top-right "3BOXES AI TRY-ON" badge ──
   169→          ctx.save();
   170→          ctx.globalAlpha = 0.88;
   171→          const badgeW = Math.max(Math.floor(w * 0.34), 120);
   172→          const badgeH = Math.max(Math.floor(h * 0.04), 28);
   173→          const badgeX = w - badgeW - 12;
   174→          const badgeY = 12;
   175→          ctx.fillStyle = '#1c1917';
   176→          ctx.beginPath();
   177→          ctx.roundRect(badgeX, badgeY, badgeW, badgeH, 6);
   178→          ctx.fill();
   179→          // Gold border
   180→          ctx.strokeStyle = 'rgba(218,165,32,0.6)';
   181→          ctx.lineWidth = 1;
   182→          ctx.beginPath();
   183→          ctx.roundRect(badgeX, badgeY, badgeW, badgeH, 6);
   184→          ctx.stroke();
   185→          // Text
   186→          ctx.fillStyle = '#daa520';
   187→          const badgeFontSize = Math.max(10, Math.floor(badgeH * 0.52));
   188→          ctx.font = `bold ${badgeFontSize}px Arial, Helvetica, sans-serif`;
   189→          ctx.textAlign = 'center';
   190→          ctx.textBaseline = 'middle';
   191→          ctx.fillText('3BOXES AI TRY-ON', badgeX + badgeW / 2, badgeY + badgeH / 2);
   192→          ctx.restore();
   193→
   194→          // ── Bottom watermark bar ──
   195→          ctx.save();
   196→          ctx.globalAlpha = 0.82;
   197→          const barH = Math.max(40, Math.floor(h * 0.06));
   198→          const barY = h - barH - 8;
   199→          ctx.fillStyle = 'rgba(28,25,23,0.75)';
   200→          ctx.beginPath();
   201→          ctx.roundRect(w * 0.08, barY, w * 0.84, barH, 8);
   202→          ctx.fill();
   203→
   204→          // Product name (gold)
   205→          const nameFontSize = Math.max(12, Math.floor(barH * 0.36));
   206→          ctx.fillStyle = '#daa520';
   207→          ctx.font = `bold ${nameFontSize}px Arial, Helvetica, sans-serif`;
   208→          ctx.textAlign = 'center';
   209→          ctx.textBaseline = 'middle';
   210→          const displayName = (productName || 'Product').substring(0, 40);
   211→          ctx.fillText(displayName, w / 2, barY + barH * 0.38);
   212→
   213→          // "3BOXES GIFTS • AI Style Preview" branding
   214→          const brandFontSize = Math.max(9, Math.floor(barH * 0.26));
   215→          ctx.globalAlpha = 0.65;
   216→          ctx.fillStyle = '#a8a29e';
   217→          ctx.font = `${brandFontSize}px Arial, Helvetica, sans-serif`;
   218→          ctx.fillText('3BOXES GIFTS \u2022 AI Style Preview', w / 2, barY + barH * 0.72);
   219→          ctx.restore();
   220→
   221→          resolve(canvas.toDataURL('image/png'));
   222→        } catch {
   223→          resolve(imageDataUrl);
   224→        }
   225→      };
   226→      img.onerror = () => resolve(imageDataUrl);
   227→      img.src = imageDataUrl;
   228→    } catch {
   229→      resolve(imageDataUrl);
   230→    }
   231→  });
