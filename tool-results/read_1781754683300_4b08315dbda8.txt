   382→function rgbToColorName(r: number, g: number, b: number): string {
   383→  // Convert RGB to HSV for better color naming
   384→  const rn = r / 255, gn = g / 255, bn = b / 255
   385→  const max = Math.max(rn, gn, bn)
   386→  const min = Math.min(rn, gn, bn)
   387→  const delta = max - min
   388→  const v = max
   389→  const s = max === 0 ? 0 : delta / max
   390→  let h = 0
   391→  if (delta !== 0) {
   392→    if (max === rn) h = ((gn - bn) / delta) % 6
   393→    else if (max === gn) h = (bn - rn) / delta + 2
   394→    else h = (rn - gn) / delta + 4
   395→    h *= 60
   396→    if (h < 0) h += 360
   397→  }
   398→
   399→  // Achromatic (grey/black/white)
   400→  if (s < 0.12) {
   401→    if (v < 0.15) return 'black'
   402→    if (v > 0.92) return 'white'
   403→    if (v < 0.4) return 'charcoal grey'
   404→    if (v < 0.65) return 'grey'
   405→    return 'silver'
   406→  }
   407→
   408→  // Bright/light/dark prefix — generous with "bright" for saturated colours
   409→  // so FLUX generates vibrant results (product photos are usually well-lit)
   410→  const lightPrefix = v > 0.6 ? 'bright ' : v < 0.25 ? 'dark ' : ''
   411→
   412→  // Hue-based naming
   413→  if (h < 15 || h >= 345) return `${lightPrefix}red`
   414→  if (h < 30) return v < 0.4 ? 'maroon' : 'red'
   415→  if (h < 45) return v < 0.4 ? 'burgundy' : 'orange-red'
   416→  if (h < 60) return `${lightPrefix}orange`
   417→  if (h < 70) return 'mustard yellow'
   418→  if (h < 85) return `${lightPrefix}yellow`
   419→  if (h < 100) return v > 0.6 ? 'lime' : 'olive'
   420→  if (h < 150) return `${lightPrefix}green`
   421→  if (h < 175) return 'emerald green'
   422→  if (h < 195) return 'teal'
   423→  if (h < 215) return 'turquoise'
   424→  if (h < 240) return `${lightPrefix}blue`
   425→  if (h < 260) return 'navy blue'
   426→  if (h < 285) return 'violet'
   427→  if (h < 310) return v > 0.6 ? 'pink' : 'purple'
   428→  if (h < 335) return v > 0.7 ? 'rose pink' : 'magenta'
   429→  return `${lightPrefix}red`
   430→}

   432→export async function extractColorsFromProductImage(imageBase64: string): Promise<string> {
   433→  if (!imageBase64 || !imageBase64.startsWith('data:image/')) return ''
   434→
   435→  try {
   436→    const raw = stripDataUrl(imageBase64)
   437→    const buf = Buffer.from(raw, 'base64')
   438→
   439→    // Use Jimp (pure-JS, no native binary — works on Vercel AND local)
   440→    const JimpModule = await import('jimp')
   441→    const Jimp = (JimpModule as any).Jimp || (JimpModule as any).default || JimpModule
   442→    const image = await Jimp.read(buf)
   443→    image.resize({ w: 32, h: 32 })
   444→
   445→    // Jimp's bitmap.data is RGBA (4 bytes per pixel)
   446→    const data = image.bitmap.data as Buffer
   447→    const pixelCount = 32 * 32
   448→
   449→    // Quantize pixels into colour buckets, weighting by SATURATION
   450→    // (vibrant product colours get higher priority than grey backgrounds)
   451→    const buckets = new Map<string, { count: number; satSum: number; r: number; g: number; b: number }>()
   452→    for (let i = 0; i < pixelCount; i++) {
   453→      const offset = i * 4
   454→      const r = data[offset], g = data[offset + 1], b = data[offset + 2]
   455→      // Skip transparent pixels
   456→      if (data[offset + 3] < 128) continue
   457→
   458→      const max = Math.max(r, g, b), min = Math.min(r, g, b)
   459→      const delta = max - min
   460→      const sat = max === 0 ? 0 : delta / max // 0..1
   461→
   462→      // Skip near-white backgrounds (common in e-commerce product photos)
   463→      if (max > 235 && delta < 15) continue
   464→      // Skip near-black
   465→      if (max < 25) continue
   466→      // Skip low-saturation greys (background, shadows, mannequin)
   467→      // Only keep pixels with saturation > 0.18 (vibrant enough to be product colour)
   468→      if (sat < 0.18) continue
   469→
   470→      // Quantize to 3 bits per channel
   471→      const key = `${r >> 5}-${g >> 5}-${b >> 5}`
   472→      const existing = buckets.get(key)
   473→      if (existing) {
   474→        existing.count++
   475→        existing.satSum += sat
   476→        existing.r += r
   477→        existing.g += g
   478→        existing.b += b
   479→      } else {
   480→        buckets.set(key, { count: 1, satSum: sat, r, g, b })
   481→      }
   482→    }
   483→
   484→    if (buckets.size === 0) {
   485→      // Fallback: if no vibrant colours found (e.g. black/white/silver product),
   486→      // re-run without the saturation filter
   487→      const fallback = new Map<string, { count: number; r: number; g: number; b: number }>()
   488→      for (let i = 0; i < pixelCount; i++) {
   489→        const offset = i * 4
   490→        const r = data[offset], g = data[offset + 1], b = data[offset + 2]
   491→        if (data[offset + 3] < 128) continue
   492→        const max = Math.max(r, g, b), min = Math.min(r, g, b)
   493→        if (max > 235 && max - min < 15) continue
   494→        if (max < 25) continue
   495→        const key = `${r >> 5}-${g >> 5}-${b >> 5}`
   496→        const existing = fallback.get(key)
   497→        if (existing) { existing.count++; existing.r += r; existing.g += g; existing.b += b }
   498→        else fallback.set(key, { count: 1, r, g, b })
   499→      }
   500→      if (fallback.size === 0) return ''
   501→      const sorted = Array.from(fallback.values()).sort((a, b) => b.count - a.count)
   502→      const names = sorted.slice(0, 2).map(bk => rgbToColorName(Math.round(bk.r / bk.count), Math.round(bk.g / bk.count), Math.round(bk.b / bk.count)))
   503→      const unique = Array.from(new Set(names))
   504→      console.log(`[virtual-tryon] Image-extracted colours (fallback): ${unique.join(', ')}`)
   505→      return unique.join(', ')
   506→    }
   507→
   508→    // Sort by (count × avg saturation) — vibrant + frequent colours win
   509→    const sorted = Array.from(buckets.values()).sort((a, b) =>
   510→      (b.count * (b.satSum / b.count)) - (a.count * (a.satSum / a.count))
   511→    )
   512→    const top = sorted.slice(0, 4)
   513→    const names = top.map(bucket => {
   514→      const avgR = Math.round(bucket.r / bucket.count)
   515→      const avgG = Math.round(bucket.g / bucket.count)
   516→      const avgB = Math.round(bucket.b / bucket.count)
   517→      return rgbToColorName(avgR, avgG, avgB)
   518→    })
   519→
   520→    // Deduplicate by BASE colour name — if "red" and "bright red" both appear,
   521→    // keep only "bright red" (the more descriptive/vibrant variant). This
   522→    // prevents FLUX from averaging two reds into a muted medium-red.
   523→    const byBase = new Map<string, string>()
   524→    for (const name of names) {
   525→      const base = name.replace(/^(bright |dark )/, '').trim()
   526→      const existing = byBase.get(base)
   527→      // Prefer "bright" variant over plain, and plain over "dark"
   528→      if (!existing) {
   529→        byBase.set(base, name)
   530→      } else if (name.startsWith('bright ') && !existing.startsWith('bright ')) {
   531→        byBase.set(base, name) // upgrade to bright
   532→      }
   533→    }
   534→    // Preserve original frequency order
   535→    const unique: string[] = []
   536→    for (const name of names) {
   537→      const base = name.replace(/^(bright |dark )/, '').trim()
   538→      const chosen = byBase.get(base)
   539→      if (chosen === name && !unique.includes(name)) {
   540→        unique.push(name)
   541→        byBase.delete(base) // only add once
   542→      }
   543→      if (unique.length >= 2) break // max 2 colours to keep the prompt focused
   544→    }
   545→    console.log(`[virtual-tryon] Image-extracted colours: ${unique.join(', ')} (from ${buckets.size} vibrant buckets)`)
   546→    return unique.join(', ')
   547→  } catch (err) {
   548→    console.log(`[virtual-tryon] Colour extraction failed: ${err instanceof Error ? err.message : String(err)}`)
   549→    return ''
   550→  }
   551→}

   575→// ── Strategy A: Direct ZAI image-edit (raw HTTP, no SDK) ───────────
   576→// We bypass the z-ai-web-dev-sdk and make direct fetch calls to the ZAI
   577→// API. This gives us:
   578→//   1. Full visibility into the raw API response (for debugging)
   579→//   2. Control over the image-URL download (the API returns a URL to a
   580→//      Chinese cloud host that can be slow/unreachable from Vercel — we
   581→//      handle download failures explicitly with a timeout)
   582→//   3. Better error messages (no more "Cannot read properties of
   583→//      undefined (reading 'map')" — we check the response shape)
   584→
   585→async function downloadZAIImage(
   586→  imageUrl: string,
   587→  timeoutMs: number,
   588→): Promise<{ buffer: Buffer; mime: string } | null> {
   589→  const controller = new AbortController()
   590→  const timeoutId = setTimeout(() => controller.abort(), timeoutMs)
   591→  try {
   592→    const res = await fetch(imageUrl, {
   593→      signal: controller.signal,
   594→      headers: { 'User-Agent': '3BOXES-VirtualTryOn/1.0', 'Accept': 'image/*,*/*;q=0.8' },
   595→    })
   596→    clearTimeout(timeoutId)
   597→    if (!res.ok) {
   598→      console.log(`[virtual-tryon] ZAI image download failed: HTTP ${res.status}`)
   599→      return null
   600→    }
   601→    const ct = res.headers.get('content-type') || 'image/png'
   602→    const mime = ct.split(';')[0].trim()
   603→    const buf = Buffer.from(await res.arrayBuffer())
   604→    if (buf.length < 3000) {
   605→      console.log(`[virtual-tryon] ZAI image download too small: ${buf.length} bytes`)
   606→      return null
   607→    }
   608→    return { buffer: buf, mime: mime.startsWith('image/') ? mime : 'image/png' }
   609→  } catch (err) {
   610→    clearTimeout(timeoutId)
   611→    console.log(`[virtual-tryon] ZAI image download error: ${err instanceof Error ? err.message : String(err)}`)
   612→    return null
   613→  }
   614→}

   616→async function callZAIImageEdit(
   617→  input: TryOnInput,
   618→  deadline: number,
   619→): Promise<{ success: boolean; imageUrl?: string; error?: string }> {
   620→  const config_obj = getZAIConfig()
   621→  if (!config_obj) {
   622→    return { success: false, error: 'ZAI config unavailable' }
   623→  }
   624→
   625→  const catConfig = getCategoryConfig(input.categorySlug, input.productName)
   626→  const prompt = buildEditPrompt(catConfig, input)
   627→
   628→  // Use edit-both if we have a product image, otherwise edit-selfie
   629→  const hasProductImage = input.productImageBase64 && input.productImageBase64.startsWith('data:image/')
   630→  const images = hasProductImage
   631→    ? [{ url: input.selfieData }, { url: input.productImageBase64 }]
   632→    : [{ url: input.selfieData }]
   633→
   634→  const strategyName = hasProductImage ? 'edit-both' : 'edit-selfie'
   635→  const remaining = Math.min(ZAI_EDIT_TIMEOUT_MS, deadline - Date.now() - 3_000)
   636→  if (remaining < 15_000) {
   637→    return { success: false, error: `insufficient time budget (${remaining}ms) for ZAI edit` }
   638→  }
   639→
   640→  console.log(`[virtual-tryon] ZAI image-edit (${strategyName}): ${catConfig.size}, timeout=${remaining}ms`)
   641→  console.log(`[virtual-tryon] Prompt (first 200): ${prompt.substring(0, 200)}...`)
   642→
   643→  // Build the request — direct HTTP call (bypasses the SDK)
   644→  const url = `${config_obj.baseUrl}/images/generations/edit`
   645→  const headers: Record<string, string> = {
   646→    'Content-Type': 'application/json',
   647→    'Authorization': `Bearer ${config_obj.apiKey}`,
   648→    'X-Z-AI-From': 'Z',
   649→  }
   650→  if (config_obj.chatId) headers['X-Chat-Id'] = config_obj.chatId
   651→  if (config_obj.userId) headers['X-User-Id'] = config_obj.userId
   652→  if (config_obj.token) headers['X-Token'] = config_obj.token
   653→
   654→  const requestBody = { prompt, images, size: catConfig.size }
   655→
   656→  const controller = new AbortController()
   657→  const timeoutId = setTimeout(() => controller.abort(), remaining)
   658→
   659→  try {
   660→    const start = Date.now()
   661→    const res = await fetch(url, {
   662→      method: 'POST',
   663→      headers,
   664→      body: JSON.stringify(requestBody),
   665→      signal: controller.signal,
   666→    })
   667→    clearTimeout(timeoutId)
   668→    const elapsed = ((Date.now() - start) / 1000).toFixed(1)
   669→
   670→    if (!res.ok) {
   671→      const errBody = await res.text().catch(() => 'unknown')
   672→      console.log(`[virtual-tryon] ZAI API HTTP ${res.status} after ${elapsed}s: ${errBody.substring(0, 300)}`)
   673→      return { success: false, error: `ZAI API HTTP ${res.status}: ${errBody.substring(0, 150)}` }
   674→    }
   675→
   676→    const result = await res.json() as any
   677→    console.log(`[virtual-tryon] ZAI API response after ${elapsed}s: keys=${Object.keys(result || {}).join(',')}`)
   678→
   679→    // The API returns { data: [{ url: "..." }] } or { data: [{ base64: "..." }] }
   680→    const item = result?.data?.[0]
   681→    if (!item) {
   682→      const raw = JSON.stringify(result).substring(0, 300)
   683→      console.log(`[virtual-tryon] ZAI response has no data[0]: ${raw}`)
   684→      return { success: false, error: `ZAI returned no image data after ${elapsed}s: ${raw.substring(0, 150)}` }
   685→    }
   686→
   687→    // Case 1: response has base64 directly
   688→    if (item.base64 && typeof item.base64 === 'string' && item.base64.length > 3000) {
   689→      let mime = 'image/png'
   690→      if (item.format === 'jpeg' || item.format === 'jpg') mime = 'image/jpeg'
   691→      else if (item.format === 'webp') mime = 'image/webp'
   692→      else {
   693→        const head = Buffer.from(item.base64.substring(0, 8), 'base64').toString('hex')
   694→        if (head.startsWith('ffd8ff')) mime = 'image/jpeg'
   695→        else if (head.startsWith('89504e47')) mime = 'image/png'
   696→        else if (head.startsWith('52494646')) mime = 'image/webp'
   697→      }
   698→      const dataUrl = `data:${mime};base64,${item.base64}`
   699→      console.log(`[virtual-tryon] ✅ ZAI ${strategyName} succeeded (base64) in ${elapsed}s (${(item.base64.length * 0.75 / 1024).toFixed(1)}KB, ${mime})`)
   700→      return { success: true, imageUrl: dataUrl }
   701→    }
   702→
   703→    // Case 2: response has a URL — download the image
   704→    if (item.url) {
   705→      console.log(`[virtual-tryon] ZAI returned URL, downloading: ${item.url.substring(0, 120)}...`)
   706→      const downloadTimeout = Math.min(20_000, deadline - Date.now() - 2_000)
   707→      if (downloadTimeout < 5_000) {
   708→        return { success: false, error: `insufficient time to download ZAI image (${downloadTimeout}ms)` }
   709→      }
   710→      const downloaded = await downloadZAIImage(item.url, downloadTimeout)
   711→      if (!downloaded) {
   712→        return { success: false, error: `ZAI image download failed after ${elapsed}s (URL unreachable from this environment)` }
   713→      }
   714→      const dataUrl = `data:${downloaded.mime};base64,${downloaded.buffer.toString('base64')}`
   715→      console.log(`[virtual-tryon] ✅ ZAI ${strategyName} succeeded (url→download) in ${elapsed}s + download (${(downloaded.buffer.length / 1024).toFixed(1)}KB, ${downloaded.mime})`)
   716→      return { success: true, imageUrl: dataUrl }
   717→    }
   718→
   719→    return { success: false, error: `ZAI response had neither base64 nor url after ${elapsed}s` }
   720→  } catch (err) {
   721→    clearTimeout(timeoutId)
   722→    const isTimeout = err instanceof DOMException && err.name === 'AbortError'
   723→    const msg = isTimeout ? `ZAI edit timed out (${remaining}ms)` : `ZAI error: ${(err as Error).message.substring(0, 150)}`
   724→    console.log(`[virtual-tryon] ZAI ${strategyName} failed: ${msg}`)
   725→    return { success: false, error: msg }
   726→  }
   727→}

   736→async function uploadToTmpfiles(buf: Buffer, timeoutMs: number): Promise<string | null> {
   737→  const boundary = '----3boxesTryon' + Math.random().toString(16).slice(2)
   738→  const header = Buffer.from(
   739→    `--${boundary}\r\nContent-Disposition: form-data; name="file"; filename="image.jpg"\r\nContent-Type: image/jpeg\r\n\r\n`
   740→  )
   741→  const footer = Buffer.from(`\r\n--${boundary}--\r\n`)
   742→  const body = Buffer.concat([header, buf, footer])
   743→
   744→  const controller = new AbortController()
   745→  const timeoutId = setTimeout(() => controller.abort(), timeoutMs)
   746→  try {
   747→    const res = await fetch('https://tmpfiles.org/api/v1/upload', {
   748→      method: 'POST',
   749→      headers: {
   750→        'Content-Type': `multipart/form-data; boundary=${boundary}`,
   751→        'User-Agent': '3BOXES-TryOn/1.0',
   752→      },
   753→      body,
   754→      signal: controller.signal,
   755→    })
   756→    clearTimeout(timeoutId)
   757→    if (!res.ok) return null
   758→    const json = (await res.json()) as { data?: { url?: string } }
   759→    const viewerUrl = json?.data?.url
   760→    if (!viewerUrl) return null
   761→    return viewerUrl.replace('tmpfiles.org/', 'tmpfiles.org/dl/')
   762→  } catch {
   763→    clearTimeout(timeoutId)
   764→    return null
   765→  }
   766→}

   768→function buildPollinationsPrompt(config: CategoryConfig, input: TryOnInput, imageColors: string): string {
   769→  // Prefer image-extracted colours (accurate) over text-extracted (from name/desc)
   770→  const textColors = extractColors(input.productName, input.productDescription, input.productTags)
   771→  const colors = imageColors || textColors
   772→  const genderWord = config.gender === 'woman' ? 'woman' : config.gender === 'man' ? 'man' : config.gender === 'child' ? 'child' : 'person'
   773→
   774→  // Build person description from selfie attributes (extracted client-side)
   775→  // This helps Pollinations generate a person that matches the user's
   776→  // skin tone and hair color, even though the exact face can't be preserved.
   777→  const personAttrs: string[] = []
   778→  if (input.skinTone) personAttrs.push(`${input.skinTone} skin`)
   779→  if (input.hairColor) personAttrs.push(`${input.hairColor} hair`)
   780→  const personDesc = personAttrs.length > 0 ? ` with ${personAttrs.join(' and ')}` : ''
   781→
   782→  // Build color prefix — put colors FIRST for maximum prominence.
   783→  // Tests show Pollinations FLUX prioritizes the first words of the prompt.
   784→  const colorPrefix = colors ? `A ${colors} ` : ''
   785→
   786→  // SHORT, FOCUSED prompt — tests confirmed that Pollinations FLUX responds
   787→  // best to concise prompts with explicit colour names. Long prompts dilute
   788→  // the colour signal and produce mismatched results.
   789→  const parts: string[] = []
   790→  parts.push(`${colorPrefix}${input.productName} worn by a ${genderWord}${personDesc}, ${config.placement}.`)
   791→  if (config.materialHint) parts.push(`Material: ${config.materialHint}.`)
   792→  if (input.productDescription) {
   793→    // Keep description very short — just key details
   794→    const desc = input.productDescription.substring(0, 120).replace(/\s+/g, ' ').trim()
   795→    if (desc) parts.push(`${desc}.`)
   796→  }
   797→  parts.push(`${config.framing}, studio lighting, photorealistic, sharp focus, fashion magazine quality.`)
   798→  parts.push(`Naturally worn with realistic folds and fit.`)
   799→  parts.push(`No sunglasses, no glasses, no hats, no extra props.`)
   800→  return parts.join(' ')
   801→}

   803→// Compress the selfie before upload (resize to max 768px, JPEG quality 82)
   804→// so tmpfiles.org upload is fast and Pollinations receives a reasonable size.
   805→async function compressSelfieForUpload(selfieData: string): Promise<Buffer> {
   806→  const sharp = await getSharp()
   807→  if (!sharp) {
   808→    // Fallback: raw buffer
   809→    const raw = stripDataUrl(selfieData)
   810→    return Buffer.from(raw, 'base64')
   811→  }
   812→  try {
   813→    const raw = stripDataUrl(selfieData)
   814→    const buf = Buffer.from(raw, 'base64')
   815→    const compressed = await sharp(buf)
   816→      .resize(768, 1024, { fit: 'inside', withoutEnlargement: true })
   817→      .jpeg({ quality: 82, progressive: true })
   818→      .toBuffer()
   819→    console.log(`[virtual-tryon] Selfie compressed: ${(buf.length / 1024).toFixed(1)}KB → ${(compressed.length / 1024).toFixed(1)}KB`)
   820→    return compressed
   821→  } catch {
   822→    const raw = stripDataUrl(selfieData)
   823→    return Buffer.from(raw, 'base64')
   824→  }

   827→async function callPollinationsWithSelfieReference(
   828→  input: TryOnInput,
   829→  deadline: number,
   830→): Promise<{ success: boolean; imageUrl?: string; error?: string; strategy?: string; debugInfo?: { extractedColors: string; promptPreview: string; selfieUploaded: boolean } }> {
   831→  const config = getCategoryConfig(input.categorySlug, input.productName)
   832→  const { width, height } = parseImageSize(config.size)
   833→
   834→  // Step 1: Extract REAL colours from the product image.
   835→  // Priority: client-extracted (canvas, most reliable) > server-extracted (jimp) > text-extracted (from name/desc)
   836→  let imageColors = input.clientProductColors || ''
   837→  if (!imageColors) {
   838→    imageColors = await extractColorsFromProductImage(input.productImageBase64)
   839→  }
   840→
   841→  // Step 2: Build the prompt with accurate, image-derived colours
   842→  const prompt = buildPollinationsPrompt(config, input, imageColors)
   843→
   844→  console.log(`[virtual-tryon] Pollinations v23: gender=${config.gender}, ${width}x${height}, imageColours="${imageColors}" (source=${input.clientProductColors ? 'client' : imageColors ? 'jimp' : 'text'})`)
   845→  console.log(`[virtual-tryon] Prompt (first 250): ${prompt.substring(0, 250)}...`)
   846→
   847→  // Step 3: Upload the SELFIE (not the product) as the ?image= reference
   848→  // This preserves the user's face/gender/features in the generated image.
   849→  let selfieUrl: string | null = null
   850→  if (Date.now() < deadline - 22_000) {
   851→    try {
   852→      const selfieBuf = await compressSelfieForUpload(input.selfieData)
   853→      selfieUrl = await uploadToTmpfiles(selfieBuf, UPLOAD_TIMEOUT_MS)
   854→      console.log(`[virtual-tryon] Selfie uploaded: ${selfieUrl ? 'yes' : 'no'}`)
   855→    } catch (err) {
   856→      console.log(`[virtual-tryon] Selfie upload failed: ${err instanceof Error ? err.message : String(err)}`)
   857→    }
   858→  }
   859→
   860→  const encoded = encodeURIComponent(prompt)
   861→  const seed = Math.floor(Math.random() * 1_000_000)
   862→  let url = `https://image.pollinations.ai/prompt/${encoded}?width=${width}&height=${height}&model=flux&nologo=true&seed=${seed}`
   863→  if (selfieUrl) {
   864→    url += `&image=${encodeURIComponent(selfieUrl)}`
   865→  }
   866→
   867→  // Retry up to 2 times on rate-limit (HTTP 402)
   868→  const MAX_RETRIES = 2
   869→  const RETRY_DELAYS_MS = [4_000, 6_000]
   870→
   871→  for (let attempt = 0; attempt <= MAX_RETRIES; attempt++) {
   872→    if (attempt > 0) {
   873→      const delay = RETRY_DELAYS_MS[Math.min(attempt - 1, RETRY_DELAYS_MS.length - 1)]
   874→      console.log(`[virtual-tryon] Pollinations retry ${attempt}/${MAX_RETRIES} after ${delay}ms...`)
   875→      await new Promise(r => setTimeout(r, delay))
   876→    }
   877→
   878→    const retrySeed = seed + attempt * 11111
   879→    const attemptUrl = url.replace(/&seed=\d+/, `&seed=${retrySeed}`)
   880→    const remaining = Math.min(POLLINATIONS_TIMEOUT_MS, deadline - Date.now() - 3_000)
   881→    if (remaining < 8_000) {
   882→      return { success: false, error: `insufficient time for Pollinations (${remaining}ms)` }
   883→    }
   884→
   885→    const controller = new AbortController()
   886→    const timeoutId = setTimeout(() => controller.abort(), remaining)
   887→
   888→    try {
   889→      const start = Date.now()
   890→      const res = await fetch(attemptUrl, {
   891→        signal: controller.signal,
   892→        headers: { 'Accept': 'image/jpeg, image/png, image/webp, */*', 'User-Agent': '3BOXES-VirtualTryOn/1.0' },
   893→      })
   894→      clearTimeout(timeoutId)
   895→      const elapsed = ((Date.now() - start) / 1000).toFixed(1)
   896→
   897→      if (res.status === 402 && attempt < MAX_RETRIES) {
   898→        console.log(`[virtual-tryon] Pollinations 402 rate-limited after ${elapsed}s — will retry`)
   899→        continue
   900→      }
   901→      if (!res.ok) {
   902→        const body = await res.text().catch(() => 'unknown')
   903→        if (attempt < MAX_RETRIES) continue
   904→        return { success: false, error: `Pollinations HTTP ${res.status}: ${body.substring(0, 100)}` }
   905→      }
   906→
   907→      const ct = res.headers.get('content-type') || ''
   908→      if (!ct.startsWith('image/')) {
   909→        if (attempt < MAX_RETRIES) continue
   910→        return { success: false, error: `Pollinations non-image response: ${ct}` }
   911→      }
   912→
   913→      const buf = Buffer.from(await res.arrayBuffer())
   914→      if (buf.length < 3000) {
   915→        if (attempt < MAX_RETRIES) continue
   916→        return { success: false, error: `Pollinations returned tiny image (${buf.length} bytes)` }
   917→      }
   918→
   919→      let mime = 'image/jpeg'
   920→      if (ct.includes('image/png')) mime = 'image/png'
   921→      else if (ct.includes('image/webp')) mime = 'image/webp'
   922→
   923→      const dataUrl = `data:${mime};base64,${buf.toString('base64')}`
   924→      console.log(`[virtual-tryon] ✅ Pollinations succeeded in ${elapsed}s (${(buf.length / 1024).toFixed(1)}KB)`)
   925→      return {
   926→        success: true,
   927→        imageUrl: dataUrl,
   928→        strategy: selfieUrl ? 'pollinations-selfie-img2img' : 'pollinations-text',
   929→        debugInfo: {
   930→          extractedColors: imageColors,
   931→          promptPreview: prompt.substring(0, 300),
   932→          selfieUploaded: !!selfieUrl,
   933→        },
   934→      }
   935→    } catch (err) {
   936→      clearTimeout(timeoutId)
   937→      const isTimeout = err instanceof DOMException && err.name === 'AbortError'
   938→      const msg = isTimeout ? `Pollinations timed out` : `Pollinations error: ${(err as Error).message.substring(0, 100)}`
   939→      console.log(`[virtual-tryon] ${msg} (attempt ${attempt + 1})`)
   940→      if (attempt < MAX_RETRIES) continue
   941→      return { success: false, error: msg }
   942→    }
   943→  }
   944→
   945→  return { success: false, error: 'Pollinations failed after all retries' }

  1000→export async function performVirtualTryOn(input: TryOnInput): Promise<TryOnResult> {
  1001→  const totalStart = Date.now()
  1002→  const totalDeadline = totalStart + TOTAL_TIMEOUT_MS
  1003→  const strategiesAttempted: string[] = []
  1004→  const strategyErrors: Record<string, string> = {}
  1005→  const isVercel = !!process.env.VERCEL
  1006→
  1007→  console.log(`[virtual-tryon] v23 start: "${input.productName}" (${input.categorySlug}) — VERCEL=${isVercel}, hasSelfie=${!!input.selfieData}, hasProductImg=${!!input.productImageBase64}`)
  1008→
  1009→  // Validate selfie
  1010→  if (!input.selfieData?.startsWith('data:image/')) {
  1011→    return {
  1012→      success: false,
  1013→      error: 'A valid selfie image is required.',
  1014→      errorCode: 'NO_SELFIE',
  1015→      elapsedMs: Date.now() - totalStart,
  1016→      debugInfo: { strategiesAttempted, strategyErrors },
  1017→    }
  1018→  }
  1019→
  1020→  // ── STRATEGY A: ZAI image-edit (edit-both) — PRIMARY on BOTH local & Vercel
  1021→  // v23: ZAI's internal-api.z.ai endpoint is publicly reachable. The
  1022→  // hardcoded config fallback ensures it works on Vercel without env vars.
  1023→  // edit-both passes BOTH the selfie AND the product image to the AI,
  1024→  // preserving the user's face/gender AND rendering the exact product.
  1025→  // Completes in 18-27s.
  1026→  if (Date.now() < totalDeadline - 18_000) {
  1027→    strategiesAttempted.push('zai-image-edit')
  1028→    console.log('[virtual-tryon] Strategy A: ZAI image-edit (edit-both) — PRIMARY')
  1029→    const result = await callZAIImageEdit(input, totalDeadline)
  1030→    if (result.success && result.imageUrl) {
  1031→      const elapsed = Date.now() - totalStart
  1032→      console.log(`[virtual-tryon] ✅ ZAI image-edit succeeded in ${(elapsed / 1000).toFixed(1)}s`)
  1033→      return {
  1034→        success: true,
  1035→        imageUrl: result.imageUrl,
  1036→        strategy: 'zai-image-edit',
  1037→        elapsedMs: elapsed,
  1038→        debugInfo: { strategiesAttempted, strategyErrors },
  1039→      }
  1040→    }
  1041→    strategyErrors['zai-image-edit'] = result.error || 'No image returned'
  1042→    console.log(`[virtual-tryon] ZAI image-edit failed: ${result.error?.substring(0, 150)}`)
  1043→  }
  1044→
  1045→  // ── STRATEGY B: Pollinations (selfie reference + image-extracted colours)
  1046→  // FALLBACK only — used when ZAI is completely unreachable (e.g. outage).
  1047→  // Pollinations FLUX does NOT preserve the user's face (?image= is ignored
  1048→  // for face preservation), so this is a degraded experience. The product
  1049→  // colours ARE accurate (extracted from the actual product image via jimp).
  1050→  if (Date.now() < totalDeadline - 12_000) {
  1051→    strategiesAttempted.push('pollinations-selfie-img2img')
  1052→    console.log('[virtual-tryon] Strategy B: Pollinations fallback (selfie reference + image-extracted colours)')
  1053→    const result = await callPollinationsWithSelfieReference(input, totalDeadline)
  1054→    if (result.success && result.imageUrl) {
  1055→      const elapsed = Date.now() - totalStart
  1056→      console.log(`[virtual-tryon] ✅ Pollinations succeeded in ${(elapsed / 1000).toFixed(1)}s`)
  1057→      return {
  1058→        success: true,
  1059→        imageUrl: result.imageUrl,
  1060→        strategy: result.strategy || 'pollinations-selfie-img2img',
  1061→        elapsedMs: elapsed,
  1062→        debugInfo: {
  1063→          strategiesAttempted,
  1064→          strategyErrors,
  1065→          extractedColors: result.debugInfo?.extractedColors,
  1066→          promptPreview: result.debugInfo?.promptPreview,
  1067→          selfieUploaded: result.debugInfo?.selfieUploaded,
  1068→        },
  1069→      }
  1070→    }
  1071→    strategyErrors['pollinations-selfie-img2img'] = result.error || 'No image returned'
  1072→    console.log(`[virtual-tryon] Pollinations failed: ${result.error?.substring(0, 150)}`)
  1073→  }
  1074→
  1075→  // ── All strategies failed ───────────────────────────────────────
  1076→  const elapsed = Date.now() - totalStart
  1077→  console.log(`[virtual-tryon] ❌ All strategies failed in ${(elapsed / 1000).toFixed(1)}s`)
  1078→  console.log(`[virtual-tryon] Strategies: ${strategiesAttempted.join(', ')}`)
  1079→  console.log(`[virtual-tryon] Errors: ${JSON.stringify(strategyErrors)}`)
  1080→
  1081→  return {
  1082→    success: false,
  1083→    error: 'We could not generate your style preview right now. Please try again in a moment.',
  1084→    errorCode: 'ALL_STRATEGIES_FAILED',
  1085→    elapsedMs: elapsed,
  1086→    debugInfo: { strategiesAttempted, strategyErrors },
  1087→  }
  1088→}
