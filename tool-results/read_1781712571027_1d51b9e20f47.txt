    78→// ── Helper: Compress image ─────────────────────────────────────────

    80→function compressImage(dataUrl: string, maxSize = 1024, quality = 0.85): Promise<string> {
    81→  return new Promise((resolve, reject) => {
    82→    const img = document.createElement('img');
    83→    img.onload = () => {
    84→      const canvas = document.createElement('canvas');
    85→      let { width, height } = img;
    86→
    87→      if (width > maxSize || height > maxSize) {
    88→        if (width > height) {
    89→          height = Math.round((height * maxSize) / width);
    90→          width = maxSize;
    91→        } else {
    92→          width = Math.round((width * maxSize) / height);
    93→          height = maxSize;
    94→        }
    95→      }
    96→
    97→      canvas.width = width;
    98→      canvas.height = height;
    99→      const ctx = canvas.getContext('2d');
   100→      if (!ctx) {
   101→        reject(new Error('Could not get canvas context'));
   102→        return;
   103→      }
   104→      ctx.drawImage(img, 0, 0, width, height);
   105→      resolve(canvas.toDataURL('image/jpeg', quality));
   106→    };
   107→    img.onerror = () => reject(new Error('Failed to load image'));
   108→    img.src = dataUrl;
   109→  });
   110→}

   112→// ── Helper: Fetch image as base64 ──────────────────────────────────
   113→
   114→async function fetchImageAsBase64(url: string): Promise<string | null> {
   115→  if (url.startsWith('data:')) return url;
   116→
   117→  const strategies: string[] = [];
   118→
   119→  if (url.startsWith('//')) {
   120→    strategies.push(`/api/image-proxy?url=${encodeURIComponent(`https:${url}`)}`);
   121→  } else if (url.startsWith('http://') || url.startsWith('https://')) {
   122→    strategies.push(`/api/image-proxy?url=${encodeURIComponent(url)}`);
   123→    strategies.push(url);
   124→  } else if (url.startsWith('/') && !url.startsWith('/api/')) {
   125→    strategies.push(url);
   126→  }
   127→
   128→  for (const fetchUrl of strategies) {
   129→    try {
   130→      const response = await fetch(fetchUrl, { signal: AbortSignal.timeout(8000) });
   131→      if (!response.ok) continue;
   132→      const contentType = response.headers.get('content-type') || '';
   133→      if (!contentType.startsWith('image/') && !contentType.startsWith('application/octet-stream')) continue;
   134→      const blob = await response.blob();
   135→      const base64 = await new Promise<string>((resolve) => {
   136→        const reader = new FileReader();
   137→        reader.onloadend = () => resolve(reader.result as string);
   138→        reader.readAsDataURL(blob);
   139→      });
   140→      if (base64.startsWith('data:')) return base64;
   141→    } catch {}
   142→  }
   143→  return null;
   144→}

   149→function add3BoxesWatermark(imageDataUrl: string, productName: string): Promise<string> {
   150→  return new Promise((resolve) => {
   151→    try {
   152→      const img = document.createElement('img');
   153→      img.crossOrigin = 'anonymous';
   154→      img.onload = () => {
   155→        try {
   156→          const canvas = document.createElement('canvas');
   157→          // Use the FULL image dimensions — never crop
   158→          canvas.width = img.naturalWidth;
   159→          canvas.height = img.naturalHeight;
   160→          const ctx = canvas.getContext('2d');
   161→          if (!ctx) {
   162→            resolve(imageDataUrl);
   163→            return;
   164→          }
   165→
   166→          // Draw the original image at FULL size
   167→          ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
   168→
   169→          const w = canvas.width;
   170→          const h = canvas.height;
   171→
   172→          // ── Top-right "3BOXES AI TRY-ON" badge ──
   173→          ctx.save();
   174→          ctx.globalAlpha = 0.88;
   175→          const badgeW = Math.max(Math.floor(w * 0.34), 120);
   176→          const badgeH = Math.max(Math.floor(h * 0.04), 28);
   177→          const badgeX = w - badgeW - 12;
   178→          const badgeY = 12;
   179→          ctx.fillStyle = '#1c1917';
   180→          ctx.beginPath();
   181→          ctx.roundRect(badgeX, badgeY, badgeW, badgeH, 6);
   182→          ctx.fill();
   183→          // Gold border
   184→          ctx.strokeStyle = 'rgba(218,165,32,0.6)';
   185→          ctx.lineWidth = 1;
   186→          ctx.beginPath();
   187→          ctx.roundRect(badgeX, badgeY, badgeW, badgeH, 6);
   188→          ctx.stroke();
   189→          // Text
   190→          ctx.fillStyle = '#daa520';
   191→          const badgeFontSize = Math.max(10, Math.floor(badgeH * 0.52));
   192→          ctx.font = `bold ${badgeFontSize}px Arial, Helvetica, sans-serif`;
   193→          ctx.textAlign = 'center';
   194→          ctx.textBaseline = 'middle';
   195→          ctx.fillText('3BOXES AI TRY-ON', badgeX + badgeW / 2, badgeY + badgeH / 2);
   196→          ctx.restore();
   197→
   198→          // ── Bottom watermark bar ──
   199→          ctx.save();
   200→          ctx.globalAlpha = 0.82;
   201→          const barH = Math.max(40, Math.floor(h * 0.06));
   202→          const barY = h - barH - 8;
   203→          ctx.fillStyle = 'rgba(28,25,23,0.75)';
   204→          ctx.beginPath();
   205→          ctx.roundRect(w * 0.08, barY, w * 0.84, barH, 8);
   206→          ctx.fill();
   207→
   208→          // Product name (gold)
   209→          const nameFontSize = Math.max(12, Math.floor(barH * 0.36));
   210→          ctx.fillStyle = '#daa520';
   211→          ctx.font = `bold ${nameFontSize}px Arial, Helvetica, sans-serif`;
   212→          ctx.textAlign = 'center';
   213→          ctx.textBaseline = 'middle';
   214→          const displayName = (productName || 'Product').substring(0, 40);
   215→          ctx.fillText(displayName, w / 2, barY + barH * 0.38);
   216→
   217→          // "3BOXES GIFTS • AI Style Preview" branding
   218→          const brandFontSize = Math.max(9, Math.floor(barH * 0.26));
   219→          ctx.globalAlpha = 0.65;
   220→          ctx.fillStyle = '#a8a29e';
   221→          ctx.font = `${brandFontSize}px Arial, Helvetica, sans-serif`;
   222→          ctx.fillText('3BOXES GIFTS \u2022 AI Style Preview', w / 2, barY + barH * 0.72);
   223→          ctx.restore();
   224→
   225→          resolve(canvas.toDataURL('image/png'));
   226→        } catch {
   227→          resolve(imageDataUrl);
   228→        }
   229→      };
   230→      img.onerror = () => resolve(imageDataUrl);
   231→      img.src = imageDataUrl;
   232→    } catch {
   233→      resolve(imageDataUrl);
   234→    }
   235→  });
   236→}
