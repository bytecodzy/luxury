    76→    76→function compressImage(dataUrl: string, maxSize = 1024, quality = 0.85): Promise<string> {
    77→    77→  return new Promise((resolve, reject) => {
    78→    78→    const img = document.createElement('img');
    79→    79→    img.onload = () => {
    80→    80→      const canvas = document.createElement('canvas');
    81→    81→      let { width, height } = img;
    82→    82→
    83→    83→      if (width > maxSize || height > maxSize) {
    84→    84→        if (width > height) {
    85→    85→          height = Math.round((height * maxSize) / width);
    86→    86→          width = maxSize;
    87→    87→        } else {
    88→    88→          width = Math.round((width * maxSize) / height);
    89→    89→          height = maxSize;
    90→    90→        }
    91→    91→      }
    92→    92→
    93→    93→      canvas.width = width;
    94→    94→      canvas.height = height;
    95→    95→      const ctx = canvas.getContext('2d');
    96→    96→      if (!ctx) {
    97→    97→        reject(new Error('Could not get canvas context'));
    98→    98→        return;
    99→    99→      }
   100→   100→      ctx.drawImage(img, 0, 0, width, height);
   101→   101→      resolve(canvas.toDataURL('image/jpeg', quality));
   102→   102→    };
   103→   103→    img.onerror = () => reject(new Error('Failed to load image'));
   104→   104→    img.src = dataUrl;
   105→   105→  });
   106→   106→}

   110→   110→async function fetchImageAsBase64(url: string): Promise<string | null> {
   111→   111→  if (url.startsWith('data:')) return url;
   112→   112→
   113→   113→  const strategies: string[] = [];
   114→   114→
   115→   115→  if (url.startsWith('//')) {
   116→   116→    strategies.push(`/api/image-proxy?url=${encodeURIComponent(`https:${url}`)}`);
   117→   117→  } else if (url.startsWith('http://') || url.startsWith('https://')) {
   118→   118→    strategies.push(`/api/image-proxy?url=${encodeURIComponent(url)}`);
   119→   119→    strategies.push(url);
   120→   120→  } else if (url.startsWith('/') && !url.startsWith('/api/')) {
   121→   121→    strategies.push(url);
   122→   122→  }
   123→   123→
   124→   124→  for (const fetchUrl of strategies) {
   125→   125→    try {
   126→   126→      const response = await fetch(fetchUrl, { signal: AbortSignal.timeout(8000) });
   127→   127→      if (!response.ok) continue;
   128→   128→      const contentType = response.headers.get('content-type') || '';
   129→   129→      if (!contentType.startsWith('image/') && !contentType.startsWith('application/octet-stream')) continue;
   130→   130→      const blob = await response.blob();
   131→   131→      const base64 = await new Promise<string>((resolve) => {
   132→   132→        const reader = new FileReader();
   133→   133→        reader.onloadend = () => resolve(reader.result as string);
   134→   134→        reader.readAsDataURL(blob);
   135→   135→      });
   136→   136→      if (base64.startsWith('data:')) return base64;
   137→   137→    } catch {}
   138→   138→  }
   139→   139→  return null;
   140→   140→}

   157→   157→          if (!ctx) {
   158→   158→            resolve(imageDataUrl);
   159→   159→            return;
   160→   160→          }

   162→   162→          // Draw the original image at FULL size
   163→   163→          ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
