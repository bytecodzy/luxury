   361→   361→// ── Strategy A: Direct ZAI image-edit ──────────────────────────────
   362→   362→
   363→   363→async function callZAIImageEdit(
   364→   364→  input: TryOnInput,
   365→   365→  deadline: number,
   366→   366→): Promise<{ success: boolean; imageUrl?: string; error?: string }> {
   367→   367→  const zai = await getZAI()
   368→   368→  if (!zai) {
   369→   369→    return { success: false, error: 'ZAI SDK unavailable (not configured or Vercel environment)' }
   370→   370→  }
   371→   371→
   372→   372→  const config = getCategoryConfig(input.categorySlug, input.productName)
   373→   373→  const prompt = buildEditPrompt(config, input)
   374→   374→
   375→   375→  // Use edit-both if we have a product image, otherwise edit-selfie
   376→   376→  const hasProductImage = input.productImageBase64 && input.productImageBase64.startsWith('data:image/')
   377→   377→  const images = hasProductImage
   378→   378→    ? [{ url: input.selfieData }, { url: input.productImageBase64 }]
   379→   379→    : [{ url: input.selfieData }]
   380→   380→
   381→   381→  const strategyName = hasProductImage ? 'edit-both' : 'edit-selfie'
   382→   382→  const remaining = Math.min(ZAI_EDIT_TIMEOUT_MS, deadline - Date.now() - 3_000)
   383→   383→  if (remaining < 15_000) {
   384→   384→    return { success: false, error: `insufficient time budget (${remaining}ms) for ZAI edit` }
   385→   385→  }
   386→   386→
   387→   387→  console.log(`[virtual-tryon] ZAI image-edit (${strategyName}): ${config.size}, timeout=${remaining}ms`)
   388→   388→  console.log(`[virtual-tryon] Prompt (first 200): ${prompt.substring(0, 200)}...`)
   389→   389→
   390→   390→  try {
   391→   391→    const start = Date.now()
   392→   392→    const response = await Promise.race([
   393→   393→      zai.images.generations.edit({
   394→   394→        prompt,
   395→   395→        images,
   396→   396→        size: config.size,
   397→   397→      }),
   398→   398→      new Promise<never>((_, rej) => setTimeout(() => rej(new Error('ZAI edit timeout')), remaining)),
   399→   399→    ])
   400→   400→    const elapsed = ((Date.now() - start) / 1000).toFixed(1)
   401→   401→
   402→   402→    const item = response?.data?.[0]
   403→   403→    const b64 = item?.base64
   404→   404→    if (!b64 || typeof b64 !== 'string' || b64.length < 3000) {
   405→   405→      return { success: false, error: `ZAI edit returned no usable image after ${elapsed}s (base64 length=${b64?.length || 0})` }
   406→   406→    }
   407→   407→
   408→   408→    // Detect format
   409→   409→    let mime = 'image/png'
   410→   410→    if (item.format === 'jpeg' || item.format === 'jpg') mime = 'image/jpeg'
   411→   411→    else if (item.format === 'webp') mime = 'image/webp'
   412→   412→    else {
   413→   413→      const head = Buffer.from(b64.substring(0, 8), 'base64').toString('hex')
   414→   414→      if (head.startsWith('ffd8ff')) mime = 'image/jpeg'
   415→   415→      else if (head.startsWith('89504e47')) mime = 'image/png'
   416→   416→      else if (head.startsWith('52494646')) mime = 'image/webp'
   417→   417→    }
   418→   418→
   419→   419→    const dataUrl = `data:${mime};base64,${b64}`
   420→   420→    const byteLen = Math.floor(b64.length * 0.75)
   421→   421→    console.log(`[virtual-tryon] ✅ ZAI ${strategyName} succeeded in ${elapsed}s (${(byteLen / 1024).toFixed(1)}KB, ${mime})`)
   422→   422→    return { success: true, imageUrl: dataUrl }
   423→   423→  } catch (err) {
   424→   424→    const msg = err instanceof Error ? err.message : String(err)
   425→   425→    console.log(`[virtual-tryon] ZAI ${strategyName} failed: ${msg.substring(0, 200)}`)
   426→   426→    return { success: false, error: msg.substring(0, 200) }
   427→   427→  }
   428→   428→}

   440→   440→    `--${boundary}\r\nContent-Disposition: form-data; name="file"; filename="image.jpg"\r\nContent-Type: image/jpeg\r\n\r\n`
   441→   441→  )
   442→   442→  const footer = Buffer.from(`\r\n--${boundary}--\r\n`)
   443→   443→  const body = Buffer.concat([header, buf, footer])
   444→   444→
   445→   445→  const controller = new AbortController()
   446→   446→  const timeoutId = setTimeout(() => controller.abort(), timeoutMs)
   447→   447→  try {
   448→   448→    const res = await fetch('https://tmpfiles.org/api/v1/upload', {
   449→   449→      method: 'POST',
   450→   450→      headers: {
   451→   451→        'Content-Type': `multipart/form-data; boundary=${boundary}`,
   452→   452→        'User-Agent': '3BOXES-TryOn/1.0',

   458→   458→    if (!res.ok) return null
   459→   459→    const json = (await res.json()) as { data?: { url?: string } }
   460→   460→    const viewerUrl = json?.data?.url
   461→   461→    if (!viewerUrl) return null
   462→   462→    return viewerUrl.replace('tmpfiles.org/', 'tmpfiles.org/dl/')
   463→   463→  } catch {

   465→   465→    return null
   466→   466→  }

   488→   488→  parts.push(`No glasses, no sunglasses, no hats, no extra props — only the product described.`)
   489→   489→  return parts.join(' ')
   490→   490→}

   492→   492→async function callPollinationsWithProductImage(
   493→   493→  input: TryOnInput,
   494→   494→  deadline: number,
   495→   495→): Promise<{ success: boolean; imageUrl?: string; error?: string; strategy?: string }> {
   496→   496→  const config = getCategoryConfig(input.categorySlug, input.productName)
   497→   497→  const prompt = buildPollinationsPrompt(config, input)
   498→   498→  const { width, height } = parseImageSize(config.size)
   499→   499→
   500→   500→  console.log(`[virtual-tryon] Pollinations: gender=${config.gender}, ${width}x${height}`)
   501→   501→  console.log(`[virtual-tryon] Prompt (first 200): ${prompt.substring(0, 200)}...`)
   502→   502→
   503→   503→  // Upload the product image so Pollinations can use it as a reference
   504→   504→  let productUrl: string | null = null
   505→   505→  if (input.productImageBase64 && Date.now() < deadline - 20_000) {
   506→   506→    try {
   507→   507→      const raw = stripDataUrl(input.productImageBase64)
   508→   508→      const buf = Buffer.from(raw, 'base64')
   509→   509→      productUrl = await uploadToTmpfiles(buf, UPLOAD_TIMEOUT_MS)
   510→   510→      console.log(`[virtual-tryon] Product image uploaded: ${productUrl ? 'yes' : 'no'}`)
   511→   511→    } catch (err) {
   512→   512→      console.log(`[virtual-tryon] Product upload failed: ${err instanceof Error ? err.message : String(err)}`)
   513→   513→    }
   514→   514→  }
   515→   515→
   516→   516→  const encoded = encodeURIComponent(prompt)
   517→   517→  const seed = Math.floor(Math.random() * 1_000_000)
   518→   518→  let url = `https://image.pollinations.ai/prompt/${encoded}?width=${width}&height=${height}&model=flux&nologo=true&seed=${seed}`
   519→   519→  if (productUrl) {
   520→   520→    url += `&image=${encodeURIComponent(productUrl)}`
   521→   521→  }
   522→   522→
   523→   523→  // Retry up to 2 times on rate-limit (HTTP 402)
   524→   524→  const MAX_RETRIES = 2
   525→   525→  const RETRY_DELAYS_MS = [4_000, 6_000]
   526→   526→
   527→   527→  for (let attempt = 0; attempt <= MAX_RETRIES; attempt++) {
   528→   528→    if (attempt > 0) {
   529→   529→      const delay = RETRY_DELAYS_MS[Math.min(attempt - 1, RETRY_DELAYS_MS.length - 1)]
   530→   530→      console.log(`[virtual-tryon] Pollinations retry ${attempt}/${MAX_RETRIES} after ${delay}ms...`)
   531→   531→      await new Promise(r => setTimeout(r, delay))
   532→   532→    }
   533→   533→
   534→   534→    const retrySeed = seed + attempt * 11111
   535→   535→    const attemptUrl = url.replace(/&seed=\d+/, `&seed=${retrySeed}`)
   536→   536→    const remaining = Math.min(POLLINATIONS_TIMEOUT_MS, deadline - Date.now() - 3_000)
   537→   537→    if (remaining < 8_000) {
   538→   538→      return { success: false, error: `insufficient time for Pollinations (${remaining}ms)` }
   539→   539→    }
   540→   540→
   541→   541→    const controller = new AbortController()
   542→   542→    const timeoutId = setTimeout(() => controller.abort(), remaining)
   543→   543→
   544→   544→    try {
   545→   545→      const start = Date.now()
   546→   546→      const res = await fetch(attemptUrl, {
   547→   547→        signal: controller.signal,
   548→   548→        headers: { 'Accept': 'image/jpeg, image/png, image/webp, */*', 'User-Agent': '3BOXES-VirtualTryOn/1.0' },
   549→   549→      })
   550→   550→      clearTimeout(timeoutId)
   551→   551→      const elapsed = ((Date.now() - start) / 1000).toFixed(1)
   552→   552→
   553→   553→      if (res.status === 402 && attempt < MAX_RETRIES) {
   554→   554→        console.log(`[virtual-tryon] Pollinations 402 rate-limited after ${elapsed}s — will retry`)
   555→   555→        continue
   556→   556→      }
   557→   557→      if (!res.ok) {
   558→   558→        const body = await res.text().catch(() => 'unknown')
   559→   559→        if (attempt < MAX_RETRIES) continue
   560→   560→        return { success: false, error: `Pollinations HTTP ${res.status}: ${body.substring(0, 100)}` }
   561→   561→      }
   562→   562→
   563→   563→      const ct = res.headers.get('content-type') || ''
   564→   564→      if (!ct.startsWith('image/')) {
   565→   565→        if (attempt < MAX_RETRIES) continue
   566→   566→        return { success: false, error: `Pollinations non-image response: ${ct}` }
   567→   567→      }
   568→   568→
   569→   569→      const buf = Buffer.from(await res.arrayBuffer())
   570→   570→      if (buf.length < 3000) {
   571→   571→        if (attempt < MAX_RETRIES) continue
   572→   572→        return { success: false, error: `Pollinations returned tiny image (${buf.length} bytes)` }
   573→   573→      }
   574→   574→
   575→   575→      let mime = 'image/jpeg'
   576→   576→      if (ct.includes('image/png')) mime = 'image/png'
   577→   577→      else if (ct.includes('image/webp')) mime = 'image/webp'
   578→   578→
   579→   579→      const dataUrl = `data:${mime};base64,${buf.toString('base64')}`
   580→   580→      console.log(`[virtual-tryon] ✅ Pollinations succeeded in ${elapsed}s (${(buf.length / 1024).toFixed(1)}KB)`)
   581→   581→      return { success: true, imageUrl: dataUrl, strategy: productUrl ? 'pollinations-product-img2img' : 'pollinations-text' }
   582→   582→    } catch (err) {
   583→   583→      clearTimeout(timeoutId)
   584→   584→      const isTimeout = err instanceof DOMException && err.name === 'AbortError'
   585→   585→      const msg = isTimeout ? `Pollinations timed out` : `Pollinations error: ${(err as Error).message.substring(0, 100)}`
   586→   586→      console.log(`[virtual-tryon] ${msg} (attempt ${attempt + 1})`)
   587→   587→      if (attempt < MAX_RETRIES) continue
   588→   588→      return { success: false, error: msg }
   589→   589→    }
   590→   590→  }
   591→   591→
   592→   592→  return { success: false, error: 'Pollinations failed after all retries' }
   593→   593→}

   650→   650→export async function performVirtualTryOn(input: TryOnInput): Promise<TryOnResult> {
   651→   651→  const totalStart = Date.now()
   652→   652→  const totalDeadline = totalStart + TOTAL_TIMEOUT_MS
   653→   653→  const strategiesAttempted: string[] = []
   654→   654→  const strategyErrors: Record<string, string> = {}
   655→   655→  const isVercel = !!process.env.VERCEL
   656→   656→
   657→   657→  console.log(`[virtual-tryon] v21 start: "${input.productName}" (${input.categorySlug}) — VERCEL=${isVercel}, hasSelfie=${!!input.selfieData}, hasProductImg=${!!input.productImageBase64}`)
   658→   658→
   659→   659→  // Validate selfie
   660→   660→  if (!input.selfieData?.startsWith('data:image/')) {
   661→   661→    return {
   662→   662→      success: false,
   663→   663→      error: 'A valid selfie image is required.',
   664→   664→      errorCode: 'NO_SELFIE',
   665→   665→      elapsedMs: Date.now() - totalStart,
   666→   666→      debugInfo: { strategiesAttempted, strategyErrors },
   667→   667→    }
   668→   668→  }
   669→   669→
   670→   670→  // ── STRATEGY A: Direct ZAI image-edit (LOCAL / SANDBOX only) ────
   671→   671→  // Calls ZAI's images.generations.edit with BOTH the selfie and the
   672→   672→  // product image (edit-both). This preserves the user's face/gender
   673→   673→  // AND renders the exact product. Completes in 20-27s.
   674→   674→  if (!isVercel && Date.now() < totalDeadline - 20_000) {
   675→   675→    strategiesAttempted.push('zai-image-edit')
   676→   676→    console.log('[virtual-tryon] Strategy A: Direct ZAI image-edit (edit-both)')
   677→   677→    const result = await callZAIImageEdit(input, totalDeadline)
   678→   678→    if (result.success && result.imageUrl) {
   679→   679→      const elapsed = Date.now() - totalStart
   680→   680→      console.log(`[virtual-tryon] ✅ ZAI image-edit succeeded in ${(elapsed / 1000).toFixed(1)}s`)
   681→   681→      return {
   682→   682→        success: true,
   683→   683→        imageUrl: result.imageUrl,
   684→   684→        strategy: 'zai-image-edit',
   685→   685→        elapsedMs: elapsed,
   686→   686→        debugInfo: { strategiesAttempted, strategyErrors },
   687→   687→      }
   688→   688→    }
   689→   689→    strategyErrors['zai-image-edit'] = result.error || 'No image returned'
   690→   690→    console.log(`[virtual-tryon] ZAI image-edit failed: ${result.error?.substring(0, 150)}`)
   691→   691→  } else if (isVercel) {
   692→   692→    strategiesAttempted.push('zai-image-edit-skipped')
   693→   693→    strategyErrors['zai-image-edit-skipped'] = 'Vercel environment — ZAI auth fails on public API'
   694→   694→    console.log('[virtual-tryon] Skipping ZAI (Vercel environment)')
   695→   695→  }
   696→   696→
   697→   697→  // ── STRATEGY B: Pollinations with product image (FALLBACK) ──────
   698→   698→  // Used on Vercel (always) and on local when ZAI fails. Uses the
   699→   699→  // PRODUCT IMAGE as the Pollinations `?image=` reference so the
   700→   700→  // generated image shows the CORRECT product. The person is a model
   701→   701→  // matching the category's gender.
   702→   702→  if (Date.now() < totalDeadline - 12_000) {
   703→   703→    strategiesAttempted.push('pollinations-product-img2img')
   704→   704→    console.log('[virtual-tryon] Strategy B: Pollinations (product image reference)')
   705→   705→    const result = await callPollinationsWithProductImage(input, totalDeadline)
   706→   706→    if (result.success && result.imageUrl) {
   707→   707→      const elapsed = Date.now() - totalStart
   708→   708→      console.log(`[virtual-tryon] ✅ Pollinations succeeded in ${(elapsed / 1000).toFixed(1)}s`)
   709→   709→      return {
   710→   710→        success: true,
   711→   711→        imageUrl: result.imageUrl,
   712→   712→        strategy: result.strategy || 'pollinations-product-img2img',
   713→   713→        elapsedMs: elapsed,
   714→   714→        debugInfo: { strategiesAttempted, strategyErrors },
   715→   715→      }
   716→   716→    }
   717→   717→    strategyErrors['pollinations-product-img2img'] = result.error || 'No image returned'
   718→   718→    console.log(`[virtual-tryon] Pollinations failed: ${result.error?.substring(0, 150)}`)
   719→   719→  }
   720→   720→
   721→   721→  // ── All strategies failed ───────────────────────────────────────
   722→   722→  const elapsed = Date.now() - totalStart
   723→   723→  console.log(`[virtual-tryon] ❌ All strategies failed in ${(elapsed / 1000).toFixed(1)}s`)
   724→   724→  console.log(`[virtual-tryon] Strategies: ${strategiesAttempted.join(', ')}`)
   725→   725→  console.log(`[virtual-tryon] Errors: ${JSON.stringify(strategyErrors)}`)
   726→   726→
   727→   727→  return {
   728→   728→    success: false,
   729→   729→    error: 'We could not generate your style preview right now. Please try again in a moment.',
   730→   730→    errorCode: 'ALL_STRATEGIES_FAILED',
   731→   731→    elapsedMs: elapsed,
   732→   732→    debugInfo: { strategiesAttempted, strategyErrors },
   733→   733→  }
   734→   734→}
