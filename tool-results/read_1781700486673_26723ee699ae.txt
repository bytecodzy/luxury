     4→ * ROOT CAUSE OF PREVIOUS MISMATCH (diagnosed in v18):
     5→ *   v17 passed the PRODUCT image as the Pollinations `?image=` reference and
     6→ *   hardcoded gender in the prompt (e.g. "a graceful Indian woman"). This

    16→ *   3. The AI now PRESERVES the user's face, gender, skin tone, body type,

    19→ *      the product name, description, tags, and category) and instructs the

    21→ *   5. The prompt is GENDER-NEUTRAL — it never hardcodes a gender. It says

    23→ *      the user actually is.

   291→function extractColors(name: string, description?: string, tags?: string[]): string {
   292→  const text = `${name} ${description || ''} ${(tags || []).join(' ')}`.toLowerCase()
   293→  const found = new Set<string>()
   294→  for (const color of COLOR_WORDS) {
   295→    if (text.includes(color)) found.add(color)
   296→  }
   297→  return Array.from(found).slice(0, 3).join(', ')
   298→}

   304→function buildProductDescription(
   305→  config: CategoryTryOnConfig,
   306→  input: TryOnInput,
   307→): string {
   308→  const colors = extractColors(input.productName, input.productDescription, input.productTags)
   309→  const parts: string[] = []
   310→
   311→  parts.push(`a ${config.garmentType}`)
   312→
   313→  if (colors) {
   314→    parts.push(`in ${colors}`)
   315→  }
   316→
   317→  // Add the product name for specificity
   318→  parts.push(`specifically "${input.productName}"`)
   319→
   320→  // Add material hint
   321→  if (config.materialHint) {
   322→    parts.push(`made of ${config.materialHint}`)
   323→  }
   324→
   325→  // Add description keywords if available (first 120 chars)
   326→  if (input.productDescription) {
   327→    const desc = input.productDescription.substring(0, 160).replace(/\s+/g, ' ').trim()
   328→    if (desc) {
   329→      parts.push(`(product details: ${desc})`)
   330→    }
   331→  }
   332→
   333→  // Add tags if available
   334→  if (input.productTags && input.productTags.length > 0) {
   335→    const relevantTags = input.productTags
   336→      .filter(t => !['new-arrival', 'featured', 'bestseller'].includes(t))
   337→      .slice(0, 5)
   338→    if (relevantTags.length > 0) {
   339→      parts.push(`with ${relevantTags.join(', ')} characteristics`)
   340→    }
   341→  }
   342→
   343→  return parts.join(' ')
   344→}

   348→function stripDataUrl(dataUrl: string): string {
   349→  const match = dataUrl.match(/^data:image\/[^;]+;base64,(.+)$/)
   350→  return match ? match[1] : dataUrl
   351→}

   353→/**
   354→ * Compress a selfie data URL into a thumbnail suitable for uploading to a
   355→ * free image host. Keeps the longest edge at 768px (enough for the AI to
   356→ * preserve face/body, small enough for fast upload + fetch).
   357→ */
   358→async function compressSelfie(selfieDataUrl: string): Promise<Buffer> {
   359→  const raw = stripDataUrl(selfieDataUrl)
   360→  const inputBuf = Buffer.from(raw, 'base64')
   361→  return sharp(inputBuf)
   362→    .resize(768, 1024, { fit: 'inside', withoutEnlargement: true })
   363→    .jpeg({ quality: 80, mozjpeg: true })
   364→    .toBuffer()
   365→}

   367→/**
   368→ * Upload a compressed selfie buffer to tmpfiles.org (free, anonymous, no auth).
   369→ * Returns a DIRECT download URL that Pollinations can fetch.
   370→ *
   371→ * Falls back to null if the upload fails — the caller will then use
   372→ * text-to-image instead of img2img.
   373→ */
   374→async function uploadToTmpfiles(buf: Buffer, timeoutMs: number): Promise<string | null> {
   375→  const boundary = '----3boxesTryon' + Math.random().toString(16).slice(2)
   376→  const filename = 'selfie.jpg'
   377→  const header = Buffer.from(
   378→    `--${boundary}\r\nContent-Disposition: form-data; name="file"; filename="${filename}"\r\nContent-Type: image/jpeg\r\n\r\n`
   379→  )
   380→  const footer = Buffer.from(`\r\n--${boundary}--\r\n`)
   381→  const body = Buffer.concat([header, buf, footer])
   382→
   383→  const controller = new AbortController()
   384→  const timeoutId = setTimeout(() => controller.abort(), timeoutMs)
   385→
   386→  try {
   387→    const start = Date.now()
   388→    const res = await fetch('https://tmpfiles.org/api/v1/upload', {
   389→      method: 'POST',
   390→      headers: {
   391→        'Content-Type': `multipart/form-data; boundary=${boundary}`,
   392→        'User-Agent': '3BOXES-TryOn/1.0',
   393→      },
   394→      body,
   395→      signal: controller.signal,
   396→    })
   397→    clearTimeout(timeoutId)
   398→    const elapsed = ((Date.now() - start) / 1000).toFixed(1)
   399→
   400→    if (!res.ok) {
   401→      console.log(`[virtual-tryon] tmpfiles upload HTTP ${res.status} after ${elapsed}s`)
   402→      return null
   403→    }
   404→
   405→    const json = (await res.json()) as { status?: string; data?: { url?: string } }
   406→    const viewerUrl = json?.data?.url
   407→    if (!viewerUrl || typeof viewerUrl !== 'string') {
   408→      console.log(`[virtual-tryon] tmpfiles returned no url after ${elapsed}s`)
   409→      return null
   410→    }
   411→
   412→    // Convert viewer URL → direct download URL
   413→    const directUrl = viewerUrl.replace('tmpfiles.org/', 'tmpfiles.org/dl/')
   414→    console.log(`[virtual-tryon] tmpfiles uploaded in ${elapsed}s → ${directUrl.substring(0, 60)}...`)
   415→    return directUrl
   416→  } catch (err) {
   417→    clearTimeout(timeoutId)
   418→    const isTimeout = err instanceof DOMException && err.name === 'AbortError'
   419→    console.log(`[virtual-tryon] tmpfiles upload failed: ${isTimeout ? 'timeout' : (err as Error).message}`)
   420→    return null
   421→  }
   422→}

   433→/**
   434→ * Build the prompt for Pollinations img2img with the SELFIE as reference.
   435→ *
   436→ * KEY DESIGN PRINCIPLES:
   437→ * 1. GENDER-NEUTRAL — never hardcode a gender. Always say "the person in the
   438→ *    reference image" so the AI uses the user's actual gender.
   439→ * 2. PRESERVE THE PERSON — explicitly instruct the AI to keep the EXACT same
   440→ *    face, skin tone, body type, hair, and gender as the reference selfie.
   441→ * 3. DESCRIBE THE PRODUCT — rich detail about what to wear/hold, extracted
   442→ *    from the product name, description, tags, and category.

   449→    `CRITICAL: Keep the EXACT same face, gender, skin tone, body type, body proportions, hair, and age as the person in the reference image. Do NOT change the person's identity or gender.`,

   467→/**
   468→ * Call Pollinations. If `referenceImageUrl` is provided (the selfie URL),
   469→ * use image-to-image so the AI preserves the person's appearance.
   470→ * Otherwise fall back to plain text-to-image.
   471→ */
   472→async function callPollinations(
   473→  prompt: string,
   474→  size: ImageSize,
   475→  timeoutMs: number,
   476→  referenceImageUrl?: string,
   477→): Promise<{ success: boolean; imageUrl?: string; error?: string }> {
   478→  const { width, height } = parseImageSize(size)
   479→  const encoded = encodeURIComponent(prompt)
   480→  const seed = Math.floor(Math.random() * 1_000_000)
   481→  // flux model gives the best photorealistic results; nologo removes branding
   482→  let url = `https://image.pollinations.ai/prompt/${encoded}?width=${width}&height=${height}&model=flux&nologo=true&seed=${seed}`
   483→  if (referenceImageUrl) {
   484→    url += `&image=${encodeURIComponent(referenceImageUrl)}`
   485→  }
   486→
   487→  const mode = referenceImageUrl ? 'selfie-img2img' : 'text'
   488→  console.log(`[virtual-tryon] Pollinations ${mode}: ${width}x${height} (timeout ${timeoutMs}ms)`)
   489→  console.log(`[virtual-tryon] Prompt (first 220): ${prompt.substring(0, 220)}...`)
   490→
   491→  const controller = new AbortController()
   492→  const timeoutId = setTimeout(() => controller.abort(), timeoutMs)
   493→
   494→  try {
   495→    const start = Date.now()
   496→    const res = await fetch(url, {
   497→      signal: controller.signal,
   498→      headers: {
   499→        'Accept': 'image/jpeg, image/png, image/webp, */*',
   500→        'User-Agent': '3BOXES-VirtualTryOn/1.0',
   501→      },
   502→    })
   503→    clearTimeout(timeoutId)
   504→    const elapsed = ((Date.now() - start) / 1000).toFixed(1)
   505→
   506→    if (!res.ok) {
   507→      const body = await res.text().catch(() => 'unknown')
   508→      return { success: false, error: `Pollinations HTTP ${res.status} after ${elapsed}s: ${body.substring(0, 150)}` }
   509→    }
   510→
   511→    const ct = res.headers.get('content-type') || ''
   512→    if (!ct.startsWith('image/')) {
   513→      const body = await res.text().catch(() => 'unknown')
   514→      return { success: false, error: `Pollinations non-image "${ct}" after ${elapsed}s: ${body.substring(0, 150)}` }
   515→    }
   516→
   517→    const buf = Buffer.from(await res.arrayBuffer())
   518→    // Tiny images (<3KB) are almost always error placeholders from Pollinations
   519→    if (buf.length < 3000) {
   520→      return { success: false, error: `Pollinations returned tiny image (${buf.length} bytes) — likely an error` }
   521→    }
   522→
   523→    let mime = 'image/jpeg'
   524→    if (ct.includes('image/png')) mime = 'image/png'
   525→    else if (ct.includes('image/webp')) mime = 'image/webp'
   526→    else {
   527→      const hex = buf.subarray(0, 4).toString('hex')
   528→      if (hex === '89504e47') mime = 'image/png'
   529→      else if (hex.startsWith('ffd8ff')) mime = 'image/jpeg'
   530→      else if (hex.startsWith('52494646')) mime = 'image/webp'
   531→    }
   532→
   533→    const dataUrl = `data:${mime};base64,${buf.toString('base64')}`
   534→    console.log(`[virtual-tryon] ✅ Pollinations ${mode} succeeded in ${elapsed}s (${(buf.length / 1024).toFixed(1)}KB, ${mime})`)
   535→    return { success: true, imageUrl: dataUrl }
   536→  } catch (err) {
   537→    clearTimeout(timeoutId)
   538→    const isTimeout = err instanceof DOMException && err.name === 'AbortError'
   539→    const msg = isTimeout
   540→      ? `Pollinations timed out after ${(timeoutMs / 1000).toFixed(0)}s`
   541→      : `Pollinations fetch error: ${(err as Error).message.substring(0, 200)}`
   542→    console.log(`[virtual-tryon] ${msg}`)
   543→    return { success: false, error: msg }
   544→  }
   545→}

   556→export async function checkIDMVTONSpaceStatus(): Promise<{ awake: boolean }> {
   557→  const now = Date.now()

   565→// ── Main Try-On Function ───────────────────────────────────────────
   566→
   567→export async function performVirtualTryOn(input: TryOnInput): Promise<TryOnResult> {
   568→  const totalStart = Date.now()
   569→  const totalDeadline = totalStart + TOTAL_TIMEOUT_MS
   570→  const config = getCategoryConfig(input.categorySlug, input.productName)
   571→  const strategiesAttempted: string[] = []
   572→  const strategyErrors: Record<string, string> = {}
   573→  const isVercel = !!process.env.VERCEL
   574→
   575→  console.log(`[virtual-tryon] v18 start: "${input.productName}" (${input.categorySlug}) — VERCEL=${isVercel}, hasSelfie=${!!input.selfieData}`)
   576→
   577→  // Validate selfie
   578→  if (!input.selfieData?.startsWith('data:image/')) {
   579→    return {
   580→      success: false,
   581→      error: 'A valid selfie image is required.',
   582→      errorCode: 'NO_SELFIE',
   583→      elapsedMs: Date.now() - totalStart,
   584→      debugInfo: { strategiesAttempted, strategyErrors },
   585→    }
   586→  }
   587→
   588→  // Build the rich product description ONCE — used by all strategies
   589→  const productDesc = buildProductDescription(config, input)
   590→  console.log(`[virtual-tryon] Product description: ${productDesc.substring(0, 200)}...`)
   591→
   592→  // ── STRATEGY 1: Pollinations SELFIE img2img (PRIMARY) ────────────
   593→  // Upload the user's selfie to tmpfiles.org, then use it as the `?image=`
   594→  // reference for Pollinations. The AI will PRESERVE the user's face,
   595→  // gender, skin tone, and body type from the selfie, and ADD the product
   596→  // described in the text prompt.
   597→  if (Date.now() < totalDeadline - 20_000) {
   598→    strategiesAttempted.push('pollinations-selfie-img2img')
   599→    console.log('[virtual-tryon] Strategy 1: Pollinations selfie img2img (person-preserving)')
   600→
   601→    // Step A: compress the selfie
   602→    let selfieBuf: Buffer | null = null
   603→    try {
   604→      selfieBuf = await compressSelfie(input.selfieData)
   605→      console.log(`[virtual-tryon] Compressed selfie → ${selfieBuf.length} bytes`)
   606→    } catch (err) {
   607→      strategyErrors['pollinations-selfie-img2img'] = `compress failed: ${(err as Error).message}`
   608→      console.log(`[virtual-tryon] Selfie compress failed: ${(err as Error).message}`)
   609→    }
   610→
   611→    // Step B: upload selfie to tmpfiles.org for a public URL
   612→    let selfieUrl: string | null = null
   613→    if (selfieBuf) {
   614→      const uploadTimeout = Math.min(UPLOAD_TIMEOUT_MS, totalDeadline - Date.now() - 15_000)
   615→      if (uploadTimeout > 3000) {
   616→        selfieUrl = await uploadToTmpfiles(selfieBuf, uploadTimeout)
   617→      }
   618→    }
   619→
   620→    // Step C: call Pollinations with the selfie as reference
   621→    if (selfieUrl) {
   622→      const prompt = buildSelfieImg2ImgPrompt(config, productDesc)
   623→      const remaining = Math.min(POLLINATIONS_TIMEOUT_MS, totalDeadline - Date.now() - 5_000)
   624→      const result = await callPollinations(prompt, config.size, remaining, selfieUrl)
   625→      if (result.success && result.imageUrl) {
   626→        const elapsed = Date.now() - totalStart
   627→        console.log(`[virtual-tryon] ✅ Selfie img2img succeeded in ${(elapsed / 1000).toFixed(1)}s`)
   628→        return { success: true, imageUrl: result.imageUrl, strategy: 'pollinations-selfie-img2img', elapsedMs: elapsed, debugInfo: { strategiesAttempted, strategyErrors } }
   629→      }
   630→      strategyErrors['pollinations-selfie-img2img'] = result.error || 'No image returned'
   631→      console.log(`[virtual-tryon] Selfie img2img failed: ${result.error?.substring(0, 120)}`)
   632→    } else if (!strategyErrors['pollinations-selfie-img2img']) {
   633→      strategyErrors['pollinations-selfie-img2img'] = 'tmpfiles selfie upload failed — no reference URL'
   634→      console.log('[virtual-tryon] No selfie URL available, falling back to text-to-image')
   635→    }
   636→  }
   637→
   638→  // ── STRATEGY 2: Pollinations TEXT-TO-IMAGE (fallback) ───────────
   639→  // Used when the selfie upload failed. Uses a detailed text prompt that
   640→  // describes both the product AND a generic person. This won't preserve
   641→  // the user's exact appearance, but it will at least show the product.
   642→  if (Date.now() < totalDeadline - 12_000) {
   643→    strategiesAttempted.push('pollinations-text')
   644→    console.log('[virtual-tryon] Strategy 2: Pollinations text-to-image (fallback)')
   645→    const prompt = buildTextPrompt(config, productDesc)
   646→    const remaining = Math.min(POLLINATIONS_TIMEOUT_MS, totalDeadline - Date.now() - 5_000)
   647→    const result = await callPollinations(prompt, config.size, remaining)
   648→    if (result.success && result.imageUrl) {
   649→      const elapsed = Date.now() - totalStart
   650→      console.log(`[virtual-tryon] ✅ Text-to-image succeeded in ${(elapsed / 1000).toFixed(1)}s`)
   651→      return { success: true, imageUrl: result.imageUrl, strategy: 'pollinations-text', elapsedMs: elapsed, debugInfo: { strategiesAttempted, strategyErrors } }
   652→    }
   653→    strategyErrors['pollinations-text'] = result.error || 'No image returned'
   654→    console.log(`[virtual-tryon] Text-to-image failed: ${result.error?.substring(0, 120)}`)
   655→  }
   656→
   657→  // ── All strategies failed ───────────────────────────────────────
   658→  const elapsed = Date.now() - totalStart
   659→  console.log(`[virtual-tryon] ❌ All strategies failed in ${(elapsed / 1000).toFixed(1)}s`)
   660→  console.log(`[virtual-tryon] Strategies: ${strategiesAttempted.join(', ')}`)
   661→  console.log(`[virtual-tryon] Errors: ${JSON.stringify(strategyErrors)}`)
   662→
   663→  // Final retry: text-to-image with a fresh seed
   664→  if (elapsed < totalDeadline - 15_000) {
   665→    console.log('[virtual-tryon] Final retry: text-to-image with fresh seed')
   666→    strategiesAttempted.push('pollinations-text-retry')
   667→    const prompt = buildTextPrompt(config, productDesc)
   668→    const retryTime = Math.min(30_000, totalDeadline - Date.now() - 3_000)
   669→    const retryResult = await callPollinations(prompt, config.size, retryTime)
   670→    if (retryResult.success && retryResult.imageUrl) {
   671→      const elapsedRetry = Date.now() - totalStart
   672→      console.log(`[virtual-tryon] ✅ Text retry succeeded in ${(elapsedRetry / 1000).toFixed(1)}s`)
   673→      return { success: true, imageUrl: retryResult.imageUrl, strategy: 'pollinations-text-retry', elapsedMs: elapsedRetry, debugInfo: { strategiesAttempted, strategyErrors } }
   674→    }
   675→    strategyErrors['pollinations-text-retry'] = retryResult.error || 'Retry failed'
   676→  }
   677→
   678→  return {
   679→    success: false,
   680→    error: 'We could not generate your style preview right now. Please check your internet connection and try again in a moment.',
   681→    errorCode: 'ALL_STRATEGIES_FAILED',
   682→    elapsedMs: elapsed,
   683→    debugInfo: { strategiesAttempted, strategyErrors },
   684→  }
   685→}
