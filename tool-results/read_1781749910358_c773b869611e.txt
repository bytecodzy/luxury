     1→/**
     2→ * Virtual Try-On Engine v22 — Direct ZAI + Pollinations (Selfie Reference + Image Colours)
     3→ *
     4→ * ─────────────────────────────────────────────────────────────────────────
     5→ *  WHY v22?
     6→ *  ─────────────────────────────────────────────────────────────────────────
     7→ *  v21 used the PRODUCT image as the Pollinations `?image=` reference on
     8→ *  Vercel. This caused TWO problems:
     9→ *    1. The user's SELFIE was never sent to Pollinations → the generated
    10→ *       person didn't match the uploaded selfie (wrong face, wrong gender
    11→ *       features, wrong hair).
    12→ *    2. The product colours came from the product NAME/DESCRIPTION text,
    13→ *       not the actual product IMAGE → the generated product had the wrong
    14→ *       colours (e.g. "maroon" from the name when the actual photo was
    15→ *       bright red).
    16→ *
    17→ *  v22 fixes BOTH issues:
    18→ *    1. Uploads the user's SELFIE as the Pollinations `?image=` reference
    19→ *       so the generated person preserves the user's face, gender, skin
    20→ *       tone, hairstyle, and features.
    21→ *    2. Extracts the REAL dominant colours from the actual product IMAGE
    22→ *       using sharp (lazy-loaded) and includes them in the prompt → the
    23→ *       generated product matches the actual product's colours.
    24→ *
    25→ *  ARCHITECTURE:
    26→ *    Client (browser) POST /api/try-on
    27→ *      └─► performVirtualTryOn()  (THIS FILE)
    28→ *            ├─► Strategy A: Direct ZAI image-edit (LOCAL / SANDBOX only)
    29→ *            │     • Explicit config from /etc/.z-ai-config
    30→ *            │     • edit-both: selfie + product image → preserves face + product
    31→ *            │     • 20-27s, high quality
    32→ *            └─► Strategy B: Pollinations selfie-img2img (VERCEL + fallback)
    33→ *                  • Extract REAL colours from product image via sharp
    34→ *                  • Upload SELFIE → use as ?image= reference
    35→ *                  • Hyper-detailed prompt with image-extracted colours
    36→ *                  • Preserves user's face/gender/features
    37→ *                  • 5-15s, always works
    38→ *
    39→ *  ENVIRONMENT BEHAVIOUR:
    40→ *    • LOCAL / SANDBOX: Direct ZAI edit-both (best quality, preserves face
    41→ *      AND product). Falls back to Pollinations selfie-img2img if ZAI down.
    42→ *    • VERCEL: Pollinations selfie-img2img (ZAI auth fails on public API).
    43→ *      The user's selfie is the reference → person matches. The product
    44→ *      colours are extracted from the actual product image → product matches.
    45→ * ─────────────────────────────────────────────────────────────────────────

   105→let cachedZAIConfig: ZAIConfig | null | undefined = undefined
   106→
   107→function getZAIConfig(): ZAIConfig | null {
   108→  if (cachedZAIConfig !== undefined) return cachedZAIConfig
   109→
   110→  // On Vercel, ZAI auth fails on the public API — skip entirely
   111→  if (process.env.VERCEL) {
   112→    cachedZAIConfig = null
   113→    return null
   114→  }
   115→
   116→  // Try env vars first
   117→  if (process.env.ZAI_BASE_URL && process.env.ZAI_API_KEY) {
   118→    cachedZAIConfig = {
   119→      baseUrl: process.env.ZAI_BASE_URL,
   120→      apiKey: process.env.ZAI_API_KEY,
   121→      chatId: process.env.ZAI_CHAT_ID || '',
   122→      token: process.env.ZAI_TOKEN || '',
   123→      userId: process.env.ZAI_USER_ID || '',
   124→    }
   125→    return cachedZAIConfig
   126→  }
   127→
   128→  // Read from config files (sandbox)
   129→  const configPaths = [
   130→    '/etc/.z-ai-config',
   131→    path.join(process.cwd(), '.z-ai-config'),
   132→    path.join(os.homedir(), '.z-ai-config'),
   133→  ]
   134→  for (const filePath of configPaths) {
   135→    try {
   136→      const configStr = fs.readFileSync(filePath, 'utf-8')
   137→      const config = JSON.parse(configStr)
   138→      if (config.baseUrl && config.apiKey) {
   139→        cachedZAIConfig = {
   140→          baseUrl: config.baseUrl,
   141→          apiKey: config.apiKey,
   142→          chatId: config.chatId || '',
   143→          token: config.token || '',
   144→          userId: config.userId || '',
   145→        }
   146→        console.log(`[virtual-tryon] ZAI config loaded from ${filePath}`)
   147→        return cachedZAIConfig
   148→      }
   149→    } catch {
   150→      // continue
   151→    }
   152→  }
   153→
   154→  cachedZAIConfig = null
   155→  return null
   156→}

   342→function extractColors(name: string, description?: string, tags?: string[]): string {
   343→  const text = `${name} ${description || ''} ${(tags || []).join(' ')}`.toLowerCase()
   344→  const found = new Set<string>()
   345→  for (const color of COLOR_WORDS) {
   346→    if (text.includes(color)) found.add(color)
   347→  }
   348→  return Array.from(found).slice(0, 3).join(', ')
   349→}

   367→function rgbToColorName(r: number, g: number, b: number): string {
   368→  // Convert RGB to HSV for better color naming
   369→  const rn = r / 255, gn = g / 255, bn = b / 255
   370→  const max = Math.max(rn, gn, bn)
   371→  const min = Math.min(rn, gn, bn)
   372→  const delta = max - min
   373→  const v = max
   374→  const s = max === 0 ? 0 : delta / max
   375→  let h = 0
   376→  if (delta !== 0) {
   377→    if (max === rn) h = ((gn - bn) / delta) % 6
   378→    else if (max === gn) h = (bn - rn) / delta + 2
   379→    else h = (rn - gn) / delta + 4
   380→    h *= 60
   381→    if (h < 0) h += 360
   382→  }
   383→
   384→  // Achromatic (grey/black/white)
   385→  if (s < 0.12) {
   386→    if (v < 0.15) return 'black'
   387→    if (v > 0.92) return 'white'
   388→    if (v < 0.4) return 'charcoal grey'
   389→    if (v < 0.65) return 'grey'
   390→    return 'silver'
   391→  }
   392→
   393→  // Bright/light/dark prefix — generous with "bright" for saturated colours
   394→  // so FLUX generates vibrant results (product photos are usually well-lit)
   395→  const lightPrefix = v > 0.6 ? 'bright ' : v < 0.25 ? 'dark ' : ''
   396→
   397→  // Hue-based naming
   398→  if (h < 15 || h >= 345) return `${lightPrefix}red`
   399→  if (h < 30) return v < 0.4 ? 'maroon' : 'red'
   400→  if (h < 45) return v < 0.4 ? 'burgundy' : 'orange-red'
   401→  if (h < 60) return `${lightPrefix}orange`
   402→  if (h < 70) return 'mustard yellow'
   403→  if (h < 85) return `${lightPrefix}yellow`
   404→  if (h < 100) return v > 0.6 ? 'lime' : 'olive'
   405→  if (h < 150) return `${lightPrefix}green`
   406→  if (h < 175) return 'emerald green'
   407→  if (h < 195) return 'teal'
   408→  if (h < 215) return 'turquoise'
   409→  if (h < 240) return `${lightPrefix}blue`
   410→  if (h < 260) return 'navy blue'
   411→  if (h < 285) return 'violet'
   412→  if (h < 310) return v > 0.6 ? 'pink' : 'purple'
   413→  if (h < 335) return v > 0.7 ? 'rose pink' : 'magenta'
   414→  return `${lightPrefix}red`
   415→}

   417→export async function extractColorsFromProductImage(imageBase64: string): Promise<string> {
   418→  if (!imageBase64 || !imageBase64.startsWith('data:image/')) return ''
   419→
   420→  try {
   421→    const raw = stripDataUrl(imageBase64)
   422→    const buf = Buffer.from(raw, 'base64')
   423→
   424→    // Use Jimp (pure-JS, no native binary — works on Vercel AND local)
   425→    const JimpModule = await import('jimp')
   426→    const Jimp = (JimpModule as any).Jimp || (JimpModule as any).default || JimpModule
   427→    const image = await Jimp.read(buf)
   428→    image.resize({ w: 32, h: 32 })
   429→
   430→    // Jimp's bitmap.data is RGBA (4 bytes per pixel)
   431→    const data = image.bitmap.data as Buffer
   432→    const pixelCount = 32 * 32
   433→
   434→    // Quantize pixels into colour buckets, weighting by SATURATION
   435→    // (vibrant product colours get higher priority than grey backgrounds)
   436→    const buckets = new Map<string, { count: number; satSum: number; r: number; g: number; b: number }>()
   437→    for (let i = 0; i < pixelCount; i++) {
   438→      const offset = i * 4
   439→      const r = data[offset], g = data[offset + 1], b = data[offset + 2]
   440→      // Skip transparent pixels
   441→      if (data[offset + 3] < 128) continue
   442→
   443→      const max = Math.max(r, g, b), min = Math.min(r, g, b)
   444→      const delta = max - min
   445→      const sat = max === 0 ? 0 : delta / max // 0..1
   446→
   447→      // Skip near-white backgrounds (common in e-commerce product photos)
   448→      if (max > 235 && delta < 15) continue
   449→      // Skip near-black
   450→      if (max < 25) continue
   451→      // Skip low-saturation greys (background, shadows, mannequin)
   452→      // Only keep pixels with saturation > 0.18 (vibrant enough to be product colour)
   453→      if (sat < 0.18) continue
   454→
   455→      // Quantize to 3 bits per channel
   456→      const key = `${r >> 5}-${g >> 5}-${b >> 5}`
   457→      const existing = buckets.get(key)
   458→      if (existing) {
   459→        existing.count++
   460→        existing.satSum += sat
   461→        existing.r += r
   462→        existing.g += g
   463→        existing.b += b
   464→      } else {
   465→        buckets.set(key, { count: 1, satSum: sat, r, g, b })
   466→      }
   467→    }
   468→
   469→    if (buckets.size === 0) {
   470→      // Fallback: if no vibrant colours found (e.g. black/white/silver product),
   471→      // re-run without the saturation filter
   472→      const fallback = new Map<string, { count: number; r: number; g: number; b: number }>()
   473→      for (let i = 0; i < pixelCount; i++) {
   474→        const offset = i * 4
   475→        const r = data[offset], g = data[offset + 1], b = data[offset + 2]
   476→        if (data[offset + 3] < 128) continue
   477→        const max = Math.max(r, g, b), min = Math.min(r, g, b)
   478→        if (max > 235 && max - min < 15) continue
   479→        if (max < 25) continue
   480→        const key = `${r >> 5}-${g >> 5}-${b >> 5}`
   481→        const existing = fallback.get(key)
   482→        if (existing) { existing.count++; existing.r += r; existing.g += g; existing.b += b }
   483→        else fallback.set(key, { count: 1, r, g, b })
   484→      }
   485→      if (fallback.size === 0) return ''
   486→      const sorted = Array.from(fallback.values()).sort((a, b) => b.count - a.count)
   487→      const names = sorted.slice(0, 2).map(bk => rgbToColorName(Math.round(bk.r / bk.count), Math.round(bk.g / bk.count), Math.round(bk.b / bk.count)))
   488→      const unique = Array.from(new Set(names))
   489→      console.log(`[virtual-tryon] Image-extracted colours (fallback): ${unique.join(', ')}`)
   490→      return unique.join(', ')
   491→    }
   492→
   493→    // Sort by (count × avg saturation) — vibrant + frequent colours win
   494→    const sorted = Array.from(buckets.values()).sort((a, b) =>
   495→      (b.count * (b.satSum / b.count)) - (a.count * (a.satSum / a.count))
   496→    )
   497→    const top = sorted.slice(0, 4)
   498→    const names = top.map(bucket => {
   499→      const avgR = Math.round(bucket.r / bucket.count)
   500→      const avgG = Math.round(bucket.g / bucket.count)
   501→      const avgB = Math.round(bucket.b / bucket.count)
   502→      return rgbToColorName(avgR, avgG, avgB)
   503→    })
   504→
   505→    // Deduplicate by BASE colour name — if "red" and "bright red" both appear,
   506→    // keep only "bright red" (the more descriptive/vibrant variant). This
   507→    // prevents FLUX from averaging two reds into a muted medium-red.
   508→    const byBase = new Map<string, string>()
   509→    for (const name of names) {
   510→      const base = name.replace(/^(bright |dark )/, '').trim()
   511→      const existing = byBase.get(base)
   512→      // Prefer "bright" variant over plain, and plain over "dark"
   513→      if (!existing) {
   514→        byBase.set(base, name)
   515→      } else if (name.startsWith('bright ') && !existing.startsWith('bright ')) {
   516→        byBase.set(base, name) // upgrade to bright
   517→      }
   518→    }
   519→    // Preserve original frequency order
   520→    const unique: string[] = []
   521→    for (const name of names) {
   522→      const base = name.replace(/^(bright |dark )/, '').trim()
   523→      const chosen = byBase.get(base)
   524→      if (chosen === name && !unique.includes(name)) {
   525→        unique.push(name)
   526→        byBase.delete(base) // only add once
   527→      }
   528→      if (unique.length >= 2) break // max 2 colours to keep the prompt focused
   529→    }

   562→async function callZAIImageEdit(
   563→  input: TryOnInput,
   564→  deadline: number,
   565→): Promise<{ success: boolean; imageUrl?: string; error?: string }> {
   566→  const zai = await getZAI()
   567→  if (!zai) {
   568→    return { success: false, error: 'ZAI SDK unavailable (not configured or Vercel environment)' }
   569→  }
   570→
   571→  const config = getCategoryConfig(input.categorySlug, input.productName)
   572→  const prompt = buildEditPrompt(config, input)
   573→
   574→  // Use edit-both if we have a product image, otherwise edit-selfie
   575→  const hasProductImage = input.productImageBase64 && input.productImageBase64.startsWith('data:image/')
   576→  const images = hasProductImage
   577→    ? [{ url: input.selfieData }, { url: input.productImageBase64 }]
   578→    : [{ url: input.selfieData }]
   579→
   580→  const strategyName = hasProductImage ? 'edit-both' : 'edit-selfie'
   581→  const remaining = Math.min(ZAI_EDIT_TIMEOUT_MS, deadline - Date.now() - 3_000)
   582→  if (remaining < 15_000) {
   583→    return { success: false, error: `insufficient time budget (${remaining}ms) for ZAI edit` }
   584→  }
   585→
   586→  console.log(`[virtual-tryon] ZAI image-edit (${strategyName}): ${config.size}, timeout=${remaining}ms`)
   587→  console.log(`[virtual-tryon] Prompt (first 200): ${prompt.substring(0, 200)}...`)
   588→
   589→  try {
   590→    const start = Date.now()
   591→    const response = await Promise.race([
   592→      zai.images.generations.edit({
   593→        prompt,
   594→        images,
   595→        size: config.size,
   596→      }),
   597→      new Promise<never>((_, rej) => setTimeout(() => rej(new Error('ZAI edit timeout')), remaining)),
   598→    ])
   599→    const elapsed = ((Date.now() - start) / 1000).toFixed(1)
   600→
   601→    const item = response?.data?.[0]
   602→    const b64 = item?.base64
   603→    if (!b64 || typeof b64 !== 'string' || b64.length < 3000) {
   604→      return { success: false, error: `ZAI edit returned no usable image after ${elapsed}s (base64 length=${b64?.length || 0})` }
   605→    }
   606→
   607→    // Detect format
   608→    let mime = 'image/png'
   609→    if (item.format === 'jpeg' || item.format === 'jpg') mime = 'image/jpeg'
   610→    else if (item.format === 'webp') mime = 'image/webp'
   611→    else {
   612→      const head = Buffer.from(b64.substring(0, 8), 'base64').toString('hex')
   613→      if (head.startsWith('ffd8ff')) mime = 'image/jpeg'
   614→      else if (head.startsWith('89504e47')) mime = 'image/png'
   615→      else if (head.startsWith('52494646')) mime = 'image/webp'
   616→    }
   617→
   618→    const dataUrl = `data:${mime};base64,${b64}`
   619→    const byteLen = Math.floor(b64.length * 0.75)
   620→    console.log(`[virtual-tryon] ✅ ZAI ${strategyName} succeeded in ${elapsed}s (${(byteLen / 1024).toFixed(1)}KB, ${mime})`)
   621→    return { success: true, imageUrl: dataUrl }
   622→  } catch (err) {
   623→    const msg = err instanceof Error ? err.message : String(err)
   624→    console.log(`[virtual-tryon] ZAI ${strategyName} failed: ${msg.substring(0, 200)}`)
   625→    return { success: false, error: msg.substring(0, 200) }
   626→  }
   627→}

   636→async function uploadToTmpfiles(buf: Buffer, timeoutMs: number): Promise<string | null> {
   637→  const boundary = '----3boxesTryon' + Math.random().toString(16).slice(2)
   638→  const header = Buffer.from(
   639→    `--${boundary}\r\nContent-Disposition: form-data; name="file"; filename="image.jpg"\r\nContent-Type: image/jpeg\r\n\r\n`
   640→  )
   641→  const footer = Buffer.from(`\r\n--${boundary}--\r\n`)
   642→  const body = Buffer.concat([header, buf, footer])
   643→
   644→  const controller = new AbortController()
   645→  const timeoutId = setTimeout(() => controller.abort(), timeoutMs)
   646→  try {
   647→    const res = await fetch('https://tmpfiles.org/api/v1/upload', {
   648→      method: 'POST',
   649→      headers: {
   650→        'Content-Type': `multipart/form-data; boundary=${boundary}`,
   651→        'User-Agent': '3BOXES-TryOn/1.0',
   652→      },
   653→      body,
   654→      signal: controller.signal,
   655→    })
   656→    clearTimeout(timeoutId)
   657→    if (!res.ok) return null
   658→    const json = (await res.json()) as { data?: { url?: string } }
   659→    const viewerUrl = json?.data?.url
   660→    if (!viewerUrl) return null
   661→    return viewerUrl.replace('tmpfiles.org/', 'tmpfiles.org/dl/')
   662→  } catch {
   663→    clearTimeout(timeoutId)
   664→    return null
   665→  }
   666→}

   679→  parts.push(`Wearing "${input.productName}".`)

   682→  if (input.productDescription) {
   683→    // Keep description very short — just key details
   684→    const desc = input.productDescription.substring(0, 120).replace(/\s+/g, ' ').trim()
   685→    if (desc) parts.push(`${desc}.`)

   688→  parts.push(`Naturally worn with realistic folds and fit.`)
   689→  parts.push(`No sunglasses, no glasses, no hats, no extra props.`)
   690→  return parts.join(' ')

   693→// Compress the selfie before upload (resize to max 768px, JPEG quality 82)
   694→// so tmpfiles.org upload is fast and Pollinations receives a reasonable size.
   695→async function compressSelfieForUpload(selfieData: string): Promise<Buffer> {
   696→  const sharp = await getSharp()
   697→  if (!sharp) {
   698→    // Fallback: raw buffer
   699→    const raw = stripDataUrl(selfieData)
   700→    return Buffer.from(raw, 'base64')
   701→  }
   702→  try {
   703→    const raw = stripDataUrl(selfieData)
   704→    const buf = Buffer.from(raw, 'base64')
   705→    const compressed = await sharp(buf)
   706→      .resize(768, 1024, { fit: 'inside', withoutEnlargement: true })
   707→      .jpeg({ quality: 82, progressive: true })
   708→      .toBuffer()
   709→    console.log(`[virtual-tryon] Selfie compressed: ${(buf.length / 1024).toFixed(1)}KB → ${(compressed.length / 1024).toFixed(1)}KB`)
   710→    return compressed
   711→  } catch {
   712→    const raw = stripDataUrl(selfieData)
   713→    return Buffer.from(raw, 'base64')
   714→  }

   717→async function callPollinationsWithSelfieReference(
   718→  input: TryOnInput,
   719→  deadline: number,
   720→): Promise<{ success: boolean; imageUrl?: string; error?: string; strategy?: string; debugInfo?: { extractedColors: string; promptPreview: string; selfieUploaded: boolean } }> {
   721→  const config = getCategoryConfig(input.categorySlug, input.productName)
   722→  const { width, height } = parseImageSize(config.size)
   723→
   724→  // Step 1: Extract REAL colours from the product image (not just text)
   725→  const imageColors = await extractColorsFromProductImage(input.productImageBase64)
   726→
   727→  // Step 2: Build the prompt with accurate, image-derived colours
   728→  const prompt = buildPollinationsPrompt(config, input, imageColors)
   729→
   730→  console.log(`[virtual-tryon] Pollinations v22: gender=${config.gender}, ${width}x${height}, imageColours="${imageColors}"`)
   731→  console.log(`[virtual-tryon] Prompt (first 250): ${prompt.substring(0, 250)}...`)
   732→
   733→  // Step 3: Upload the SELFIE (not the product) as the ?image= reference
   734→  // This preserves the user's face/gender/features in the generated image.
   735→  let selfieUrl: string | null = null
   736→  if (Date.now() < deadline - 22_000) {
   737→    try {
   738→      const selfieBuf = await compressSelfieForUpload(input.selfieData)
   739→      selfieUrl = await uploadToTmpfiles(selfieBuf, UPLOAD_TIMEOUT_MS)
   740→      console.log(`[virtual-tryon] Selfie uploaded: ${selfieUrl ? 'yes' : 'no'}`)
   741→    } catch (err) {
   742→      console.log(`[virtual-tryon] Selfie upload failed: ${err instanceof Error ? err.message : String(err)}`)
   743→    }
   744→  }
   745→
   746→  const encoded = encodeURIComponent(prompt)
   747→  const seed = Math.floor(Math.random() * 1_000_000)
   748→  let url = `https://image.pollinations.ai/prompt/${encoded}?width=${width}&height=${height}&model=flux&nologo=true&seed=${seed}`
   749→  if (selfieUrl) {
   750→    url += `&image=${encodeURIComponent(selfieUrl)}`
   751→  }
   752→
   753→  // Retry up to 2 times on rate-limit (HTTP 402)
   754→  const MAX_RETRIES = 2
   755→  const RETRY_DELAYS_MS = [4_000, 6_000]
   756→
   757→  for (let attempt = 0; attempt <= MAX_RETRIES; attempt++) {
   758→    if (attempt > 0) {
   759→      const delay = RETRY_DELAYS_MS[Math.min(attempt - 1, RETRY_DELAYS_MS.length - 1)]
   760→      console.log(`[virtual-tryon] Pollinations retry ${attempt}/${MAX_RETRIES} after ${delay}ms...`)
   761→      await new Promise(r => setTimeout(r, delay))
   762→    }
   763→
   764→    const retrySeed = seed + attempt * 11111
   765→    const attemptUrl = url.replace(/&seed=\d+/, `&seed=${retrySeed}`)
   766→    const remaining = Math.min(POLLINATIONS_TIMEOUT_MS, deadline - Date.now() - 3_000)
   767→    if (remaining < 8_000) {
   768→      return { success: false, error: `insufficient time for Pollinations (${remaining}ms)` }
   769→    }
   770→
   771→    const controller = new AbortController()
   772→    const timeoutId = setTimeout(() => controller.abort(), remaining)
   773→
   774→    try {
   775→      const start = Date.now()
   776→      const res = await fetch(attemptUrl, {
   777→        signal: controller.signal,
   778→        headers: { 'Accept': 'image/jpeg, image/png, image/webp, */*', 'User-Agent': '3BOXES-VirtualTryOn/1.0' },
   779→      })
   780→      clearTimeout(timeoutId)
   781→      const elapsed = ((Date.now() - start) / 1000).toFixed(1)
   782→
   783→      if (res.status === 402 && attempt < MAX_RETRIES) {
   784→        console.log(`[virtual-tryon] Pollinations 402 rate-limited after ${elapsed}s — will retry`)
   785→        continue
   786→      }
   787→      if (!res.ok) {
   788→        const body = await res.text().catch(() => 'unknown')
   789→        if (attempt < MAX_RETRIES) continue
   790→        return { success: false, error: `Pollinations HTTP ${res.status}: ${body.substring(0, 100)}` }
   791→      }
   792→
   793→      const ct = res.headers.get('content-type') || ''
   794→      if (!ct.startsWith('image/')) {
   795→        if (attempt < MAX_RETRIES) continue
   796→        return { success: false, error: `Pollinations non-image response: ${ct}` }
   797→      }
   798→
   799→      const buf = Buffer.from(await res.arrayBuffer())
   800→      if (buf.length < 3000) {
   801→        if (attempt < MAX_RETRIES) continue
   802→        return { success: false, error: `Pollinations returned tiny image (${buf.length} bytes)` }
   803→      }
   804→
   805→      let mime = 'image/jpeg'
   806→      if (ct.includes('image/png')) mime = 'image/png'
   807→      else if (ct.includes('image/webp')) mime = 'image/webp'
   808→
   809→      const dataUrl = `data:${mime};base64,${buf.toString('base64')}`
   810→      console.log(`[virtual-tryon] ✅ Pollinations succeeded in ${elapsed}s (${(buf.length / 1024).toFixed(1)}KB)`)
   811→      return {
   812→        success: true,
   813→        imageUrl: dataUrl,
   814→        strategy: selfieUrl ? 'pollinations-selfie-img2img' : 'pollinations-text',
   815→        debugInfo: {
   816→          extractedColors: imageColors,
   817→          promptPreview: prompt.substring(0, 300),
   818→          selfieUploaded: !!selfieUrl,
   819→        },
   820→      }
   821→    } catch (err) {
   822→      clearTimeout(timeoutId)
   823→      const isTimeout = err instanceof DOMException && err.name === 'AbortError'
   824→      const msg = isTimeout ? `Pollinations timed out` : `Pollinations error: ${(err as Error).message.substring(0, 100)}`
   825→      console.log(`[virtual-tryon] ${msg} (attempt ${attempt + 1})`)
   826→      if (attempt < MAX_RETRIES) continue
   827→      return { success: false, error: msg }
   828→    }
   829→  }
   830→
   831→  return { success: false, error: 'Pollinations failed after all retries' }
   832→}

   964→    console.log(`[virtual-tryon] Pollinations failed: ${result.error?.substring(0, 150)}`)
   965→  }
